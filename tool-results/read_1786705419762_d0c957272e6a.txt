   504→   504→              (m) => m.name.toLowerCase().contains(type.toLowerCase()),
   505→   505→            );
   506→   506→    if (activeManager != null) {
   507→   507→      switch (contentType) {
   508→   508→        case ItemType.anime:
   509→   509→          return activeManager.supportsAnime;
   510→   510→        case ItemType.manga:
   511→   511→          return activeManager.supportsManga;
   512→   512→        case ItemType.novel:
   513→   513→          return activeManager.supportsNovel;
   514→   514→      }
   515→   515→    }
   516→   516→    final lowerType = type.toLowerCase();
   517→   517→    switch (contentType) {
   518→   518→      case ItemType.anime:
   519→   519→        return lowerType == 'mangayomi' ||
   520→   520→            lowerType == 'aniyomi' ||
   521→   521→            lowerType == 'cloudstream' ||
   522→   522→            lowerType == 'sora';
   523→   523→      case ItemType.manga:
   524→   524→        return lowerType == 'mangayomi' ||
   525→   525→            lowerType == 'aniyomi' ||
   526→   526→            lowerType == 'sora' ||
   527→   527→            lowerType == 'kotatsu';
   528→   528→      case ItemType.novel:
   529→   529→        return lowerType == 'mangayomi' || lowerType == 'sora';
   530→   530→    }
   531→   531→  }

   672→   672→  bool _typeRequiresPlugin(String type) {
   673→   673→    if (type == 'all') return false;
   674→   674→    final activeManager =
   675→   675→        Get.find<ExtensionManager>().managers.firstWhereOrNull(
   676→   676→              (m) => m.name.toLowerCase().contains(type.toLowerCase()),
   677→   677→            );

   679→   679→    return type == 'Aniyomi' || type == 'Cloudstream';
