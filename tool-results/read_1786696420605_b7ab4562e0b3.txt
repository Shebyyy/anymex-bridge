   299→  return r.json();

   312→    if(d.ok){token=d.token;localStorage.setItem('bridge_admin_token',token);showApp()}
   313→    else{const el=document.getElementById('loginError');el.style.display='block';setTimeout(()=>el.style.display='none',2000)}
   314→  }catch(e){toast('Connection failed','error')}

   487→function baseName(name){
   488→  return (name||'').replace(/\s*[\-–—(].*$/, '').replace(/\s+(All|Multi|English|Japanese|Spanish|Portuguese|French|Italian|German|Arabic|Hindi|Thai|Vietnamese|Indonesian|Korean|Chinese|Turkish|Polish|Russian)$/i, '').trim();

   491→function populateSourceDropdown(){
   492→  const sel=document.getElementById('invokeSource');
   493→  if(!_sources.length){
   494→    sel.innerHTML='<option value="">-- No sources loaded --</option>';
   495→    return;
   496→  }
   497→  // Group by type, then by base name
   498→  const byType={};
   499→  _sources.forEach(s=>{
   500→    const t=s.type||'other';
   501→    if(!byType[t])byType[t]=[];
   502→    byType[t].push(s);
   503→  });
   504→  let html='<option value="">-- Select a source --</option>';
   505→  for(const[type,sources] of Object.entries(byType)){
   506→    html+=`<optgroup label="${type}">`;
   507→    // Group by base name
   508→    const groups=new Map();
   509→    sources.forEach(s=>{
   510→      const bn=baseName(s.name);
   511→      if(!groups.has(bn))groups.set(bn,[]);
   512→      groups.get(bn).push(s);
   513→    });
   514→    // Sort base names
   515→    [...groups.keys()].sort().forEach(bn=>{
   516→      const items=groups.get(bn);
   517→      if(items.length===1){
   518→        const s=items[0];
   519→        html+=`<option value="${s.id}" data-type="${s.type}" data-pkg="${esc(s.pkg)}">${esc(s.name)}</option>`;
   520→      }else{
   521→        // Multiple languages — use optgroup
   522→        html+=`<optgroup label="${esc(bn)}">`;
   523→        items.sort((a,b)=>(a.lang||'').localeCompare(b.lang||'')).forEach(s=>{
   524→          const label=s.lang?s.lang:s.name.replace(bn,'').replace(/^[\s\-–—(]+|[\s\-–—)]+$/g,'')||'?';
   525→          html+=`<option value="${s.id}" data-type="${s.type}" data-pkg="${esc(s.pkg)}">${esc(s.name)}</option>`;
   526→        });
   527→        html+='</optgroup>';
   528→      }
   529→    });
   530→    html+='</optgroup>';
   531→  }
   532→  sel.innerHTML=html;
   533→}
