   179→   179→  Future<PluginRelease?> fetchLatestRelease() async {
   180→   180→    try {
   181→   181→      final response = await http.get(
   182→   182→        Uri.parse(_latestReleaseUrl),
   183→   183→        headers: const {'Accept': 'application/vnd.github+json'},
   184→   184→      );
   185→   185→
   186→   186→      if (response.statusCode != 200) {
   187→   187→        Logger.e(
   188→   188→          'Failed to fetch runtime host release: ${response.statusCode}',
   189→   189→        );
   190→   190→        return null;
   191→   191→      }
   192→   192→
   193→   193→      final Map<String, dynamic> json =
   194→   194→          jsonDecode(response.body) as Map<String, dynamic>;
   195→   195→      final assets = (json['assets'] as List<dynamic>? ?? [])
   196→   196→          .whereType<Map<String, dynamic>>()
   197→   197→          .toList();
   198→   198→
   199→   199→      final extension = Platform.isAndroid ? '.apk' : '.jar';
   200→   200→      final assetJson = assets.firstWhereOrNull(
   201→   201→        (asset) =>
   202→   202→            (asset['name'] as String? ?? '').toLowerCase().endsWith(extension),
   203→   203→      );
   204→   204→
   205→   205→      if (assetJson == null) {
   206→   206→        Logger.e('Runtime host release has no compatible asset.');
   207→   207→        return null;
   208→   208→      }
   209→   209→
   210→   210→      return PluginRelease(
   211→   211→        tagName: (json['tag_name'] as String? ?? '').trim(),
   212→   212→        title: ((json['name'] as String?)?.trim().isNotEmpty ?? false)
   213→   213→            ? (json['name'] as String).trim()
   214→   214→            : (json['tag_name'] as String? ?? 'Latest Plugin Release').trim(),
   215→   215→        body: (json['body'] as String? ?? '').trim(),
   216→   216→        publishedAt: DateTime.tryParse(json['published_at'] as String? ?? ''),
   217→   217→        asset: PluginAsset(
   218→   218→          name: (assetJson['name'] as String? ?? 'anymex_bridge').trim(),
   219→   219→          downloadUrl:
   220→   220→              (assetJson['browser_download_url'] as String? ?? '').trim(),
   221→   221→          sizeBytes: (assetJson['size'] as num?)?.toInt() ?? 0,
   222→   222→        ),
   223→   223→      );
   224→   224→    } catch (error, stackTrace) {
   225→   225→      Logger.e(
   226→   226→        'Error fetching runtime host release',
   227→   227→        error: error,
   228→   228→        stackTrace: stackTrace,
   229→   229→      );
   230→   230→      return null;
   231→   231→    }
   232→   232→  }

   234→   234→  Future<List<PluginRelease>> fetchReleases() async {
   235→   235→    try {
   236→   236→      final response = await http.get(
   237→   237→        Uri.parse('https://api.github.com/repos/RyanYuuki/AnymeXExtensionRuntimeBridge/releases'),
   238→   238→        headers: const {'Accept': 'application/vnd.github+json'},
   239→   239→      );
   240→   240→      if (response.statusCode != 200) return [];
   241→   241→      final List<dynamic> jsonList = jsonDecode(response.body) as List<dynamic>;
   242→   242→      final List<PluginRelease> releases = [];
   243→   243→      for (final json in jsonList) {
   244→   244→        final assets = (json['assets'] as List<dynamic>? ?? [])
   245→   245→            .whereType<Map<String, dynamic>>()
   246→   246→            .toList();
   247→   247→        final extension = Platform.isAndroid ? '.apk' : '.jar';
   248→   248→        final assetJson = assets.firstWhereOrNull(
   249→   249→          (asset) => (asset['name'] as String? ?? '').toLowerCase().endsWith(extension),
   250→   250→        );
   251→   251→        if (assetJson != null) {
   252→   252→          releases.add(PluginRelease(
   253→   253→            tagName: (json['tag_name'] as String? ?? '').trim(),
   254→   254→            title: ((json['name'] as String?)?.trim().isNotEmpty ?? false)
   255→   255→                ? (json['name'] as String).trim()
   256→   256→                : (json['tag_name'] as String? ?? 'Plugin Release').trim(),
   257→   257→            body: (json['body'] as String? ?? '').trim(),
   258→   258→            publishedAt: DateTime.tryParse(json['published_at'] as String? ?? ''),
   259→   259→            asset: PluginAsset(
   260→   260→              name: (assetJson['name'] as String? ?? 'anymex_bridge').trim(),
   261→   261→              downloadUrl: (assetJson['browser_download_url'] as String? ?? '').trim(),
   262→   262→              sizeBytes: (assetJson['size'] as num?)?.toInt() ?? 0,
   263→   263→            ),
   264→   264→          ));
   265→   265→        }
   266→   266→      }
   267→   267→      return releases;
   268→   268→    } catch (_) {
   269→   269→      return [];
   270→   270→    }
   271→   271→  }

   273→   273→  bool isNewerVersion(String installed, String latest) {
   274→   274→    final installedParts = _normalizeVersion(installed);
   275→   275→    final latestParts = _normalizeVersion(latest);
   276→   276→    final maxLength = installedParts.length > latestParts.length
   277→   277→        ? installedParts.length
   278→   278→        : latestParts.length;
   279→   279→
   280→   280→    for (var index = 0; index < maxLength; index++) {
   281→   281→      final installedPart =
   282→   282→          index < installedParts.length ? installedParts[index] : 0;
   283→   283→      final latestPart = index < latestParts.length ? latestParts[index] : 0;
   284→   284→
   285→   285→      if (latestPart > installedPart) return true;
   286→   286→      if (latestPart < installedPart) return false;
   287→   287→    }
   288→   288→
   289→   289→    return false;
   290→   290→  }
   291→   291→
   292→   292→  List<int> _normalizeVersion(String version) {
   293→   293→    final cleaned = version.toLowerCase().replaceFirst(RegExp(r'^v'), '');
   294→   294→    final stablePart = cleaned.split('-').first;
   295→   295→    return stablePart
   296→   296→        .split('.')
   297→   297→        .map((part) => int.tryParse(part) ?? 0)
   298→   298→        .toList();
   299→   299→  }

   305→   305→  Future<bool> syncLocalApk(String apkPath) async {
   306→   306→    final isJar = apkPath.toLowerCase().endsWith('.jar');
   307→   307→    final isApk = apkPath.toLowerCase().endsWith('.apk');
   308→   308→
   309→   309→    if (Platform.isAndroid && !isApk) {
   310→   310→      errorSnackBar('Please select a valid APK file.');
   311→   311→      return false;
   312→   312→    }
   313→   313→    if (!Platform.isAndroid && !isJar) {
   314→   314→      errorSnackBar('Please select a valid JAR file.');
   315→   315→      return false;
   316→   316→    }
   317→   317→
   318→   318→    if (!await File(apkPath).exists()) {
   319→   319→      errorSnackBar('Local file not found at: $apkPath');
   320→   320→      return false;
   321→   321→    }
   322→   322→
   323→   323→    try {
   324→   324→      if (Platform.isAndroid) {
   325→   325→        await AnymeXRuntimeBridge.useLocalApk(apkPath);
   326→   326→      } else {
   327→   327→        final paths = RuntimePaths();
   328→   328→        final destPath = await paths.bridgePath;
   329→   329→        final destFile = File(destPath);
   330→   330→        if (await destFile.exists()) {
   331→   331→          await destFile.delete();
   332→   332→        }
   333→   333→        await File(apkPath).copy(destPath);
   334→   334→        
   335→   335→        final toolsDir = await paths.toolsDir;
   336→   336→        final metadataFile = File('${toolsDir.path}/metadata.json');
   337→   337→        await metadataFile.writeAsString(jsonEncode({
   338→   338→          'version': 'Local-${DateTime.now().millisecondsSinceEpoch}',
   339→   339→          'title': 'Local Synced Jar',
   340→   340→        }));
   341→   341→        await AnymeXRuntimeBridge.loadMetadata();
   342→   342→      }
   343→   343→
   344→   344→      final bridge = AnymeXRuntimeBridge.controller;
   345→   345→      if (Platform.isAndroid && bridge.isReady.value) {
   346→   346→        await Get.find<ExtensionManager>()
   347→   347→            .onRuntimeBridgeInitialization(force: true);
   348→   348→        successSnackBar('Plugin synced from local APK.');
   349→   349→        return true;
   350→   350→      } else if (!Platform.isAndroid) {
   351→   351→        successSnackBar('Plugin jar copied. Please restart the app.');
   352→   352→        return true;
   353→   353→      }
   354→   354→
   355→   355→      errorSnackBar('Sync completed, but the plugin is not ready yet.');
   356→   356→      return false;
   357→   357→    } catch (e) {
   358→   358→      errorSnackBar('Sync failed: $e');
   359→   359→      return false;
   360→   360→    }
   361→   361→  }
   362→   362→
   363→   363→  Future<bool> forceSyncLocalApk() async {

   366→   366→      errorSnackBar('Local APK not found at: $localPath');
   367→   367→      return false;

   370→   370→    return syncLocalApk(localPath);
   371→   371→  }

   426→   426→  void _checkBridgeReady() {

   429→   429→      if (_release != null) {

   444→   444→      final fetched = await widget.manager.fetchLatestRelease();

   451→   451→          _checkBridgeReady();

   456→   456→      }
   457→   457→    } catch (e) {
   458→   458→      if (mounted) {
   459→   459→        setState(() {
   460→   460→          _isLoadingRelease = false;
   461→   461→        });
   462→   462→        errorSnackBar('Error: $e');
   463→   463→        Navigator.of(context).pop();
   464→   464→      }
   465→   465→    }

   474→   474→  Future<void> _downloadAndInstall() async {
   475→   475→    if (_release == null) return;
   476→   476→    final bridge = AnymeXRuntimeBridge.controller;
   477→   477→    if (bridge.isDownloading.value) return;
   478→   478→
   479→   479→    try {
   480→   480→      await AnymeXRuntimeBridge.setupRuntime(
   481→   481→        force: widget.mode == _PluginSheetMode.update,
   482→   482→      );
   483→   483→
   484→   484→      if (bridge.isReady.value) {
   485→   485→        await Get.find<ExtensionManager>()
   486→   486→            .onRuntimeBridgeInitialization(force: true);
   487→   487→        widget.manager.persistInstalledRelease(_release!);
   488→   488→        if (mounted) {
   489→   489→          setState(() {
   490→   490→            _installFinished = true;
   491→   491→          });
   492→   492→          successSnackBar(
   493→   493→            widget.mode == _PluginSheetMode.install
   494→   494→                ? 'Plugin installed successfully.'
   495→   495→                : 'Plugin updated successfully.',
   496→   496→          );
   497→   497→        }
   498→   498→      }
   499→   499→    } catch (error) {
   500→   500→      if (mounted) errorSnackBar('Plugin installation failed: $error');
   501→   501→    }
   502→   502→  }

   504→   504→  String _formatBytes(int bytes) {
   505→   505→    if (bytes <= 0) return 'Unknown';
   506→   506→    const kb = 1024;
   507→   507→    const mb = kb * 1024;
   508→   508→    if (bytes < kb) return '$bytes B';
   509→   509→    if (bytes < mb) return '${(bytes / kb).toStringAsFixed(1)} KB';
   510→   510→    return '${(bytes / mb).toStringAsFixed(2)} MB';
   511→   511→  }
