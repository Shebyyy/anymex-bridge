    73→    73→      _showInstallPopup();

    97→    97→  void _checkForUpdates() async {
    98→    98→    setState(() => _isCheckingUpdate = true);

   100→   100→      final release = await _pluginManager.fetchLatestRelease();

   106→   106→      final currentVersion = _installedVersion;
   107→   107→      if (currentVersion.isEmpty) {
   108→   108→        _showInstallPopup();
   109→   109→        return;
   110→   110→      }
   111→   111→      if (_pluginManager.isNewerVersion(currentVersion, release.tagName)) {
   112→   112→        _showUpdatePopup();
   113→   113→      } else {
   114→   114→        successSnackBar('Plugin is already up to date.');
   115→   115→      }

   117→   117→      if (mounted) setState(() => _isCheckingUpdate = false);
   118→   118→    }

   121→   121→  void _syncLocalApk() async {
   122→   122→    if (_isSyncingLocalApk) return;
   123→   123→
   124→   124→    final result = await FilePicker.platform.pickFiles(
   125→   125→      type: FileType.custom,
   126→   126→      allowedExtensions: Platform.isAndroid ? const ['apk'] : const ['jar'],
   127→   127→      allowMultiple: false,
   128→   128→      withData: false,
   129→   129→    );
   130→   130→
   131→   131→    final apkPath = result?.files.single.path;
   132→   132→    if (apkPath == null || apkPath.isEmpty) return;
   133→   133→    if (!mounted) return;
   134→   134→
   135→   135→    setState(() => _isSyncingLocalApk = true);
   136→   136→    try {
   137→   137→      final oldVersion = _installedVersion;
   138→   138→      final synced = await _pluginManager.syncLocalApk(apkPath);
   139→   139→      if (mounted && synced) {
   140→   140→        _checkStorageStatus();
   141→   141→        if (_installedVersion != oldVersion) {
   142→   142→          setState(() {
   143→   143→            _needsRestart = true;
   144→   144→          });
   145→   145→        } else {
   146→   146→          setState(() {});
   147→   147→        }
   148→   148→      }
   149→   149→    } finally {
   150→   150→      if (mounted) setState(() => _isSyncingLocalApk = false);
   151→   151→    }

   286→   286→  void _performRollback(PluginRelease release) async {
   287→   287→    final bridge = AnymeXRuntimeBridge.controller;
   288→   288→    if (bridge.isDownloading.value) return;
   289→   289→    try {
   290→   290→      await AnymeXRuntimeBridge.setupRuntime(
   291→   291→        customDownloadUrl: release.asset.downloadUrl,
   292→   292→        force: true,
   293→   293→      );
   294→   294→      if (bridge.isReady.value) {
   295→   295→        await Get.find<ExtensionManager>()
   296→   296→            .onRuntimeBridgeInitialization(force: true);
   297→   297→        _pluginManager.persistInstalledRelease(release);
   298→   298→        _checkStorageStatus();
   299→   299→        if (mounted) {
   300→   300→          setState(() {
   301→   301→            _needsRestart = true;
   302→   302→          });
   303→   303→          successSnackBar('Rollback to ${release.tagName} successful. Restart app to apply.');
   304→   304→        }
   305→   305→      }
   306→   306→    } catch (error) {
   307→   307→      if (mounted) errorSnackBar('Rollback failed: $error');
   308→   308→    }
   309→   309→  }

   689→   689→  List<Widget> _buildExtensionSettings(BuildContext context) {
   690→   690→    final colors = Theme.of(context).colorScheme;
   691→   691→    final em = Get.find<ExtensionManager>();
   692→   692→    final settingsList = <Widget>[];
   693→   693→
   694→   694→    for (final manager in em.managers) {
   695→   695→      final managerSettings = manager.settings;
   696→   696→      if (managerSettings == null || managerSettings.isEmpty) continue;
   697→   697→
   698→   698→      settingsList.add(const SizedBox(height: 25));
   699→   699→      settingsList.add(
   700→   700→        Padding(
   701→   701→          padding: const EdgeInsets.only(left: 4.0, bottom: 10.0),
   702→   702→          child: Text(
   703→   703→            '${manager.name.toUpperCase()} SETTINGS',
   704→   704→            style: TextStyle(
   705→   705→              fontSize: 12,
   706→   706→              fontWeight: FontWeight.bold,
   707→   707→              color: colors.primary,
   708→   708→              letterSpacing: 1.2,
   709→   709→            ),
   710→   710→          ),
   711→   711→        ),
   712→   712→      );
   713→   713→
   714→   714→      final children = <Widget>[];
   715→   715→      for (final entry in managerSettings.entries) {
   716→   716→        final setting = entry.value;
   717→   717→
   718→   718→        if (setting.type == 'bool') {
   719→   719→          children.add(
   720→   720→            CustomSwitchTile(
   721→   721→              icon: Icons.settings_input_component_rounded,
   722→   722→              title: setting.label,
   723→   723→              description: setting.description,
   724→   724→              switchValue: setting.value as bool,
   725→   725→              onChanged: (val) {
   726→   726→                setting.onChanged(val);
   727→   727→                setState(() {});
   728→   728→              },
   729→   729→            ),
   730→   730→          );
   731→   731→        } else if (setting.type == 'string') {
   732→   732→          children.add(
   733→   733→            CustomTile(
   734→   734→              icon: Icons.folder_open_rounded,
   735→   735→              title: setting.label,
   736→   736→              description: (setting.value as String).isNotEmpty
   737→   737→                  ? setting.value as String
   738→   738→                  : setting.description,
   739→   739→              onTap: () => _showTextInputDialog(
   740→   740→                context,
   741→   741→                title: setting.label,
   742→   742→                initialValue: setting.value as String,
   743→   743→                onSave: (val) {
   744→   744→                  setting.onChanged(val);
   745→   745→                  setState(() {});
   746→   746→                },
   747→   747→              ),
   748→   748→            ),
   749→   749→          );
   750→   750→        }
   751→   751→      }
   752→   752→
   753→   753→      if (children.isNotEmpty) {
   754→   754→        settingsList.add(
   755→   755→          Container(
   756→   756→            decoration: BoxDecoration(
   757→   757→              color: colors.surfaceContainer.withValues(alpha: 0.3),
   758→   758→              borderRadius: BorderRadius.circular(18),
   759→   759→              border: Border.all(
   760→   760→                  color: colors.outlineVariant.withValues(alpha: 0.4)),
   761→   761→            ),
   762→   762→            child: Column(children: children),
   763→   763→          ),
   764→   764→        );
   765→   765→      }
   766→   766→    }
   767→   767→
   768→   768→    return settingsList;
   769→   769→  }

   771→   771→  void _showTextInputDialog(
   772→   772→    BuildContext context, {
   773→   773→    required String title,
   774→   774→    required String initialValue,
   775→   775→    required ValueChanged<String> onSave,
   776→   776→  }) {
   777→   777→    final controller = TextEditingController(text: initialValue);
   778→   778→    showDialog(
   779→   779→      context: context,
   780→   780→      builder: (context) {
   781→   781→        return AlertDialog(
   782→   782→          title: Text(title),
   783→   783→          content: TextField(
   784→   784→            controller: controller,
   785→   785→            decoration: const InputDecoration(
   786→   786→              border: OutlineInputBorder(),
   787→   787→              labelText: 'Enter value',
   788→   788→            ),
   789→   789→          ),
   790→   790→          actions: [
   791→   791→            TextButton(
   792→   792→              onPressed: () => Navigator.of(context).pop(),
   793→   793→              child: const Text('Cancel'),
   794→   794→            ),
   795→   795→            TextButton(
   796→   796→              onPressed: () {
   797→   797→                onSave(controller.text);
   798→   798→                Navigator.of(context).pop();
   799→   799→              },
   800→   800→              child: const Text('Save'),
   801→   801→            ),
   802→   802→          ],
   803→   803→        );
   804→   804→      },
   805→   805→    );
   806→   806→  }
