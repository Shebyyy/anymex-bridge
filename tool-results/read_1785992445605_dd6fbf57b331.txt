    28→// ─── Helper: format uptime ─────────────────────────────
    29→function formatUptime(seconds: number): string {
    30→  const d = Math.floor(seconds / 86400)
    31→  const h = Math.floor((seconds % 86400) / 3600)
    32→  const m = Math.floor((seconds % 3600) / 60)
    33→  if (d > 0) return `${d}d ${h}h ${m}m`
    34→  if (h > 0) return `${h}h ${m}m`
    35→  return `${m}m`
    36→}

   490→  const [urlInput, setUrlInput] = useState('')
   491→  const [connecting, setConnecting] = useState(false)

   493→  // Search state

   495→  const [searchSourceId, setSearchSourceId] = useState('')

   498→
   499→  // Load saved config on mount
   500→  const loadConfig = useCallback(async () => {
   501→    try {
   502→      const res = await fetch('/api/bridge-config')
   503→      const data = await res.json()
   504→      if (data.success && data.data) {
   505→        if (data.data.serverUrl) {
   506→          store.setServerUrl(data.data.serverUrl)
   507→          setUrlInput(data.data.serverUrl)
   508→          store.setIsConnected(true)
   509→        }
   510→      }
   511→    } catch {
   512→      // Config not available yet
   513→    }
   514→  }, [store])
   515→
   516→  // Fetch health data
   517→  const fetchHealth = useCallback(async () => {
   518→    if (!store.isConnected) return
   519→    store.setHealthLoading(true)
   520→    try {
   521→      const res = await fetch('/api/bridge/api/health')
   522→      const data = await res.json()
   523→      if (data.success && data.data) {
   524→        store.setHealth(data.data)
   525→      } else {
   526→        store.setHealth(null)
   527→      }
   528→    } catch {
   529→      store.setHealth(null)
   530→    } finally {
   531→      store.setHealthLoading(false)
   532→    }
   533→  }, [store.isConnected, store])
   534→
   535→  // Fetch extensions
