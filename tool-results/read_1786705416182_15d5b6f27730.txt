   483→  String? _getManagerIcon(String type) {
   484→    switch (type) {
   485→      case 'Mangayomi':
   486→        return MSource(id: '', name: '', lang: '').managerIcon;
   487→      case 'Aniyomi':
   488→        return ASource(id: '', name: '', lang: '').managerIcon;
   489→      case 'Cloudstream':
   490→        return CloudStreamSource(id: '', name: '', lang: '').managerIcon;
   491→      case 'Sora':
   492→        return SSource(id: '', name: '', lang: '').managerIcon;
   493→      case 'Kotatsu':
   494→        return KotatsuSource(id: '', name: '', lang: '').managerIcon;
   495→      default:
   496→        return null;
   497→    }
   498→  }
   499→
   500→  bool _supportsContentType(String type, ItemType contentType) {
   501→    if (type == 'all') return true;
   502→    final activeManager =
   503→        Get.find<ExtensionManager>().managers.firstWhereOrNull(
   504→              (m) => m.name.toLowerCase().contains(type.toLowerCase()),
   505→            );
   506→    if (activeManager != null) {
   507→      switch (contentType) {
   508→        case ItemType.anime:
   509→          return activeManager.supportsAnime;
   510→        case ItemType.manga:
   511→          return activeManager.supportsManga;
   512→        case ItemType.novel:
   513→          return activeManager.supportsNovel;
   514→      }
   515→    }
   516→    final lowerType = type.toLowerCase();
   517→    switch (contentType) {
   518→      case ItemType.anime:
   519→        return lowerType == 'mangayomi' ||
   520→            lowerType == 'aniyomi' ||
   521→            lowerType == 'cloudstream' ||
   522→            lowerType == 'sora';
   523→      case ItemType.manga:
   524→        return lowerType == 'mangayomi' ||
   525→            lowerType == 'aniyomi' ||
   526→            lowerType == 'sora' ||
   527→            lowerType == 'kotatsu';
   528→      case ItemType.novel:
   529→        return lowerType == 'mangayomi' || lowerType == 'sora';
   530→    }
   531→  }

   533→  Widget _buildSourceTypeChips() {
   534→    // All source types available on all platforms (iOS now uses JNI bridge)
   535→    final allSourceTypes = ['all', 'Mangayomi', 'Aniyomi', 'Cloudstream', 'Sora', 'Kotatsu'];
   536→
   537→    return Obx(() {
   538→      final selectedType = _selectedSourceType.value;
   539→      final contentType = _selectedContentType.value;
   540→      final filteredSourceTypes = allSourceTypes
   541→          .where((type) => _supportsContentType(type, contentType))
   542→          .toList();
   543→
   544→      return Container(
   545→        height: 38,
   546→        margin: const EdgeInsets.only(top: 8, bottom: 8),
   547→        child: ListView.builder(
   548→          scrollDirection: Axis.horizontal,
   549→          physics: const BouncingScrollPhysics(),
   550→          padding: const EdgeInsets.symmetric(horizontal: 16),
   551→          itemCount: filteredSourceTypes.length,
   552→          itemBuilder: (context, index) {
   553→            final type = filteredSourceTypes[index];
   554→            final needsPlugin = _typeRequiresPlugin(type) && !_isPluginInstalled;
   555→            final isSelected = !needsPlugin && selectedType == type;
   556→            final label = type == 'all' ? 'All' : type;
   557→            final iconUrl = _getManagerIcon(type);
   558→
   559→            return Opacity(
   560→              opacity: needsPlugin ? 0.5 : 1.0,
   561→              child: Padding(
   562→                padding: const EdgeInsets.only(right: 8.0),
   563→                child: _buildSourceChip(
   564→                  label: label,
   565→                  iconUrl: iconUrl,
   566→                  isSelected: isSelected,
   567→                  onTap: needsPlugin
   568→                      ? () {
   569→                          navigate(() => const SettingsExtensionManager());
   570→                        }
   571→                      : () {
   572→                          _selectedSourceType.value = type;
   573→                        },
   574→                ),
   575→              ),
   576→            );
   577→          },
   578→        ),
   579→      );
   580→    });
   581→  }

   672→  bool _typeRequiresPlugin(String type) {
   673→    if (type == 'all') return false;
   674→    final activeManager =
   675→        Get.find<ExtensionManager>().managers.firstWhereOrNull(
   676→              (m) => m.name.toLowerCase().contains(type.toLowerCase()),
   677→            );
   678→    if (activeManager != null) return activeManager.requiresPlugin;
   679→    return type == 'Aniyomi' || type == 'Cloudstream';
   680→  }
