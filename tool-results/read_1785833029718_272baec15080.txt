   132→   132→    private fun loadAnymeXRuntimeHost(apkPath: String, settingsMap: Map<String, Any?>? = null): Boolean {
   133→   133→        Log.i(TAG, "loadAnymeXRuntimeHost called with: $apkPath")
   134→   134→        val ctx = context ?: run {
   135→   135→            Log.e(TAG, "loadAnymeXRuntimeHost: context is null")
   136→   136→            return false
   137→   137→        }
   138→   138→
   139→   139→        return try {
   140→   140→            runtimeBridge = null
   141→   141→            bridgeClass = null
   142→   142→
   143→   143→            val originalApk = File(apkPath)
   144→   144→            if (!originalApk.exists()) {
   145→   145→                Log.e(TAG, "APK does not exist at path: $apkPath")
   146→   146→                return false
   147→   147→            }
   148→   148→
   149→   149→            val cacheApkName = "anymex_runtime_${originalApk.length()}_${originalApk.lastModified()}.apk"
   150→   150→            val cacheApk = File(ctx.filesDir, cacheApkName)
   151→   151→
   152→   152→            if (!cacheApk.exists()) {
   153→   153→                Log.i(TAG, "Creating new cached APK: $cacheApkName")
   154→   154→                ctx.filesDir.listFiles()?.forEach { file ->
   155→   155→                    if (file.name.startsWith("anymex_runtime_") && file.name.endsWith(".apk") && file.name != cacheApkName) {
   156→   156→                        Log.d(TAG, "Deleting old cached APK: ${file.name}")
   157→   157→                        file.delete()
   158→   158→                    }
   159→   159→                }
   160→   160→
   161→   161→                originalApk.inputStream().use { input ->
   162→   162→                    FileOutputStream(cacheApk).use { output ->
   163→   163→                        input.copyTo(output)
   164→   164→                    }
   165→   165→                }
   166→   166→                cacheApk.setReadOnly()
   167→   167→            } else {
   168→   168→                Log.i(TAG, "Using existing cached APK: $cacheApkName")
   169→   169→            }
   170→   170→
   171→   171→            ctx.cacheDir.listFiles()?.forEach { file ->
   172→   172→                if (file.isDirectory && (file.name.startsWith("anymex_dex_") || file.name.startsWith("anymex_libs_"))) {
   173→   173→                    file.deleteRecursively()
   174→   174→                }
   175→   175→            }
   176→   176→
   177→   177→            val optimizedDir = File(ctx.cacheDir, "anymex_dex_${System.currentTimeMillis()}")
   178→   178→            optimizedDir.mkdirs()
   179→   179→
   180→   180→            val libsDir = File(ctx.cacheDir, "anymex_libs_${System.currentTimeMillis()}")
   181→   181→            libsDir.mkdirs()
   182→   182→
   183→   183→            try {
   184→   184→                java.util.zip.ZipFile(cacheApk).use { zip ->
   185→   185→                    val abisList = android.os.Build.SUPPORTED_ABIS
   186→   186→                    var selectedAbi: String? = null
   187→   187→                    
   188→   188→                    for (abi in abisList) {
   189→   189→                        val prefix = "lib/$abi/"
   190→   190→                        var found = false
   191→   191→                        val entriesEnum = zip.entries()
   192→   192→                        while (entriesEnum.hasMoreElements()) {
   193→   193→                            val entry = entriesEnum.nextElement()
   194→   194→                            if (entry.name.startsWith(prefix) && entry.name.endsWith(".so")) {
   195→   195→                                found = true
   196→   196→                                break
   197→   197→                            }
   198→   198→                        }
   199→   199→                        if (found) {
   200→   200→                            selectedAbi = abi
   201→   201→                            break
   202→   202→                        }
   203→   203→                    }
   204→   204→
   205→   205→                    if (selectedAbi != null) {
   206→   206→                        Log.i(TAG, "Extracting native libraries for ABI: $selectedAbi")
   207→   207→                        val prefix = "lib/$selectedAbi/"
   208→   208→                        val entriesEnum = zip.entries()
   209→   209→                        while (entriesEnum.hasMoreElements()) {
   210→   210→                            val entry = entriesEnum.nextElement()
   211→   211→                            if (entry.name.startsWith(prefix) && entry.name.endsWith(".so")) {
   212→   212→                                val libName = entry.name.substringAfterLast('/')
   213→   213→                                val outFile = File(libsDir, libName)
   214→   214→                                zip.getInputStream(entry).use { input ->
   215→   215→                                    FileOutputStream(outFile).use { output ->
   216→   216→                                        input.copyTo(output)
   217→   217→                                    }
   218→   218→                                }
   219→   219→                                Log.d(TAG, "Extracted native library: $libName to ${outFile.absolutePath}")
   220→   220→                            }
   221→   221→                        }
   222→   222→                    } else {
   223→   223→                        Log.w(TAG, "No matching native libraries found in APK for supported ABIs: ${abisList.joinToString()}")
   224→   224→                    }
   225→   225→                }
   226→   226→            } catch (e: Exception) {
   227→   227→                Log.e(TAG, "Failed to extract native libraries: ${e.message}", e)
   228→   228→            }
   229→   229→
   230→   230→            val loader = ChildFirstClassLoader(
   231→   231→                cacheApk.absolutePath,
   232→   232→                optimizedDir.absolutePath,
   233→   233→                libsDir.absolutePath,
   234→   234→                ctx.classLoader!!
   235→   235→            )
   236→   236→
   237→   237→            bridgeClass = loader.loadClass("com.anymex.runtimehost.RuntimeBridge")
   238→   238→            Log.d(TAG, "bridgeClass loaded: $bridgeClass")
   239→   239→            runtimeBridge = bridgeClass!!.getField("INSTANCE").get(null)
   240→   240→            
   241→   241→            try {
   242→   242→                val loggerClass = loader.loadClass("com.anymex.runtimehost.Logger")
   243→   243→                val setLogCallbackMethod = loggerClass.getMethod("setLogCallback", Any::class.java, Method::class.java)
   244→   244→                val ourLogMethod = AnymexExtensionRuntimeBridgePlugin::class.java.getMethod(
   245→   245→                    "logFromHost",
   246→   246→                    String::class.java,
   247→   247→                    String::class.java,
   248→   248→                    String::class.java
   249→   249→                )
   250→   250→                setLogCallbackMethod.invoke(null, this, ourLogMethod)
   251→   251→                Log.i(TAG, "Logger callback registered successfully.")
   252→   252→            } catch (e: Throwable) {
   253→   253→                Log.e(TAG, "Failed to register Logger callback: ${e.message}")
   254→   254→            }
   255→   255→
   256→   256→            Log.i(TAG, "AnymeX Runtime Bridge initialized successfully")
   257→   257→            
   258→   258→            try {
   259→   259→                call("initialize", ctx, settingsMap)
   260→   260→            } catch (e: Throwable) {
   261→   261→                Log.e(TAG, "Failed to initialize RuntimeBridge: ${e.message}")
   262→   262→                logToFlutter("ERROR", "BRIDGE_INIT", "Failed to initialize RuntimeBridge: ${e.message}\n${Log.getStackTraceString(e)}")
   263→   263→            }
   264→   264→
   265→   265→            Log.i(TAG, "Runtime Host loaded successfully")
   266→   266→            true
   267→   267→        } catch (e: Throwable) {
   268→   268→            Log.e(TAG, "Failed to load Runtime Host APK: ${e.message}")
   269→   269→            logToFlutter("ERROR", "BRIDGE_LOAD", "Failed to load Runtime Host APK: ${e.message}\n${Log.getStackTraceString(e)}")
   270→   270→            false
   271→   271→        }
   272→   272→    }

   274→   274→    private fun handleAnymeX(call: MethodCall, result: MethodResult) {
   275→   275→        when (call.method) {
   276→   276→            "getAbi" -> {
   277→   277→                val primaryAbi = android.os.Build.SUPPORTED_ABIS.firstOrNull() ?: "arm64-v8a"
   278→   278→                result.success(primaryAbi)
   279→   279→            }
   280→   280→            "loadAnymeXRuntimeHost" -> {
   281→   281→                val path = call.argument<String>("path")
   282→   282→                val settingsMap = call.argument<Map<String, Any?>>("settings")
   283→   283→                if (path.isNullOrBlank()) {
   284→   284→                    result.error("INVALID_ARG", "path is required", null)
   285→   285→                    return
   286→   286→                }
   287→   287→                scope.launch {
   288→   288→                    try {
   289→   289→                        val ok = loadAnymeXRuntimeHost(path, settingsMap)
   290→   290→                        withContext(Dispatchers.Main) {
   291→   291→                            if (ok) result.success(true)
   292→   292→                            else result.error("LOAD_FAILED", "Failed to load runtime host APK", null)
   293→   293→                        }
   294→   294→                    } catch (e: Throwable) {
   295→   295→                        sendError(result, "loadAnymeXRuntimeHost", e)
   296→   296→                    }
   297→   297→                }
   298→   298→            }
   299→   299→            "isLoaded" -> {
   300→   300→                Log.d(TAG, "isLoaded check: runtimeBridge=$runtimeBridge, bridgeClass=$bridgeClass")
   301→   301→                result.success(runtimeBridge != null)
   302→   302→            }
   303→   303→            "cancelRequest" -> {
   304→   304→                val token = call.argument<String>("token")
   305→   305→                if (token != null) {
   306→   306→                    val ok = call("cancelRequest", token) as? Boolean ?: false
   307→   307→                    result.success(ok)
   308→   308→                } else {
   309→   309→                    result.error("INVALID_ARG", "token is required", null)
   310→   310→                }
   311→   311→            }
   312→   312→            "setCookies" -> {
   313→   313→                val url = call.argument<String>("url")
   314→   314→                val cookieString = call.argument<String>("cookieString")
   315→   315→                if (url.isNullOrBlank() || cookieString == null) {
   316→   316→                    result.error("INVALID_ARG", "url and cookieString are required", null)
   317→   317→                    return
   318→   318→                }
   319→   319→                try {
   320→   320→                    val mgr = android.webkit.CookieManager.getInstance()
   321→   321→                    mgr.setAcceptCookie(true)
   322→   322→                    cookieString.split(";").map { it.trim() }.filter { it.isNotEmpty() }.forEach { cookie ->
   323→   323→                        mgr.setCookie(url, cookie)
   324→   324→                    }
   325→   325→                    mgr.flush()
   326→   326→                    Log.d(TAG, "setCookies: applied ${cookieString.split(";").size} cookie(s) for $url")
   327→   327→                    result.success(null)
   328→   328→                } catch (e: Exception) {
   329→   329→                    Log.e(TAG, "setCookies failed: ${e.message}")
   330→   330→                    result.error("COOKIE_ERROR", e.message, null)
   331→   331→                }
   332→   332→            }
   333→   333→            "setUserAgent" -> {
   334→   334→                val url = call.argument<String>("url")
   335→   335→                val userAgent = call.argument<String>("userAgent")
   336→   336→                if (url.isNullOrBlank() || userAgent.isNullOrBlank()) {
   337→   337→                    result.error("INVALID_ARG", "url and userAgent are required", null)
   338→   338→                    return
   339→   339→                }
   340→   340→                try {
   341→   341→                    val host = android.net.Uri.parse(url).host ?: url
   342→   342→                    System.setProperty("anymex.ua.$host", userAgent)
   343→   343→                    Log.d(TAG, "setUserAgent: stored UA for host=$host")
   344→   344→                    result.success(null)
   345→   345→                } catch (e: Exception) {
   346→   346→                    Log.e(TAG, "setUserAgent failed: ${e.message}")
   347→   347→                    result.error("UA_ERROR", e.message, null)
   348→   348→                }
   349→   349→            }
   350→   350→            else -> result.notImplemented()
   351→   351→
   352→   352→        }
   353→   353→    }

   637→   637→    private fun call(methodName: String, vararg args: Any?): Any? {
   638→   638→        val bridge = runtimeBridge ?: throw IllegalStateException("Runtime Host not loaded")
   639→   639→        val cls = bridgeClass ?: throw IllegalStateException("Runtime Host class not loaded")
   640→   640→
   641→   641→        val method = cls.methods.filter { it.name == methodName }
   642→   642→            .firstOrNull { it.parameterTypes.size == args.size }
   643→   643→            ?: cls.methods.firstOrNull { it.name == methodName }
   644→   644→            ?: throw NoSuchMethodException("No method '$methodName' in RuntimeBridge")
   645→   645→
   646→   646→        val effectiveArgs = if (method.parameterTypes.size != args.size) {
   647→   647→            logToFlutter("WARNING", "BRIDGE", "Argument count mismatch for $methodName. Expected ${method.parameterTypes.size}, got ${args.size}. Adjusting.")
   648→   648→            if (args.size > method.parameterTypes.size) {
   649→   649→                args.take(method.parameterTypes.size).toTypedArray()
   650→   650→            } else {
   651→   651→                val padded = args.toMutableList()
   652→   652→                while (padded.size < method.parameterTypes.size) padded.add(null)
   653→   653→                padded.toTypedArray()
   654→   654→            }
   655→   655→        } else {
   656→   656→            args
   657→   657→        }
   658→   658→
   659→   659→        logToFlutter("INFO", "BRIDGE", "Calling Method: RuntimeBridge.$methodName")
   660→   660→        val result = method.invoke(bridge, *effectiveArgs)
   661→   661→        logToFlutter("INFO", "BRIDGE", "Method '$methodName' completed successfully")
   662→   662→
   663→   663→        return result
   664→   664→    }

   749→   749→            val packageInfo = pm.getPackageArchiveInfo(apkPath, 0)
   750→   750→                ?: throw IllegalArgumentException("Invalid APK file at $apkPath")
   751→   751→            val packageName = packageInfo.packageName
   752→   752→
   753→   753→            val dirName = if (isAnime) "exts" else "exts_manga"
   754→   754→            val privateDir = File(context.filesDir, dirName)
   755→   755→            if (!privateDir.exists()) {

   757→   757→            }
   758→   758→
   759→   759→            val srcFile = File(apkPath)
   760→   760→            val dstFile = File(privateDir, "$packageName.apk")
   761→   761→            val tmpFile = File(privateDir, "$packageName.apk.tmp")
   762→   762→
   763→   763→            srcFile.inputStream().use { input ->
   764→   764→                FileOutputStream(tmpFile).use { output ->
   765→   765→                    input.copyTo(output)
   766→   766→                }
   767→   767→            }
   768→   768→
   769→   769→            if (dstFile.exists()) dstFile.delete()
   770→   770→
   771→   771→            if (tmpFile.renameTo(dstFile)) {
   772→   772→                try {
   773→   773→                    dstFile.setReadOnly()

   775→   775→                Log.i(TAG, "Successfully installed internal extension: $packageName to ${dstFile.absolutePath}")

   777→   777→            } else {
   778→   778→                tmpFile.delete()
   779→   779→                false
   780→   780→            }
   781→   781→        } catch (e: Exception) {
   782→   782→            Log.e(TAG, "Failed to install source internally: ${e.message}", e)

   807→   807→    private class ChildFirstClassLoader(
   808→   808→        dexPath: String,
   809→   809→        optimizedDirectory: String?,
   810→   810→        librarySearchPath: String?,
   811→   811→        parent: ClassLoader
   812→   812→    ) : DexClassLoader(dexPath, optimizedDirectory, librarySearchPath, parent) {
   813→   813→
   814→   814→        private val systemClassLoader: ClassLoader? = getSystemClassLoader()
   815→   815→
   816→   816→        private fun shouldDelegateToParent(name: String?): Boolean {
   817→   817→            if (name == null) return false
   818→   818→            return name.startsWith("androidx.")
   819→   819→        }
   820→   820→
   821→   821→        override fun loadClass(name: String?, resolve: Boolean): Class<*> {
   822→   822→            var c = findLoadedClass(name)
   823→   823→
   824→   824→            if (c == null && systemClassLoader != null) {
   825→   825→                try {
   826→   826→                    c = systemClassLoader.loadClass(name)
   827→   827→                } catch (_: ClassNotFoundException) {}
   828→   828→            }
   829→   829→
   830→   830→            if (c == null && shouldDelegateToParent(name)) {
   831→   831→                try {
   832→   832→                    c = parent.loadClass(name)
   833→   833→                } catch (_: ClassNotFoundException) {}
   834→   834→            }
   835→   835→
   836→   836→            if (c == null) {
   837→   837→                try {
   838→   838→                    c = findClass(name)
   839→   839→                } catch (_: ClassNotFoundException) {
   840→   840→                    c = super.loadClass(name, resolve)
   841→   841→                }
   842→   842→            }
   843→   843→
   844→   844→            if (resolve) {
   845→   845→                resolveClass(c)
   846→   846→            }
   847→   847→
   848→   848→            return c
   849→   849→        }
   850→   850→    }
