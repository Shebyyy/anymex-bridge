   177→  Future<PluginRelease?> fetchLatestRelease() async {
   178→    try {
   179→      final response = await http.get(
   180→        Uri.parse(_latestReleaseUrl),
   181→        headers: const {'Accept': 'application/vnd.github+json'},
   182→      );
   183→
   184→      if (response.statusCode != 200) {
   185→        Logger.e(
   186→          'Failed to fetch runtime host release: ${response.statusCode}',
   187→        );
   188→        return null;
   189→      }
   190→
   191→      final Map<String, dynamic> json =
   192→          jsonDecode(response.body) as Map<String, dynamic>;
   193→      final assets = (json['assets'] as List<dynamic>? ?? [])
   194→          .whereType<Map<String, dynamic>>()
   195→          .toList();
   196→
   197→      final extension = Platform.isAndroid ? '.apk' : '.jar';
   198→      final assetJson = assets.firstWhereOrNull(
   199→        (asset) =>
   200→            (asset['name'] as String? ?? '').toLowerCase().endsWith(extension),
   201→      );
   202→
   203→      if (assetJson == null) {
   204→        Logger.e('Runtime host release has no compatible asset.');
   205→        return null;
   206→      }
   207→
   208→      return PluginRelease(
   209→        tagName: (json['tag_name'] as String? ?? '').trim(),
   210→        title: ((json['name'] as String?)?.trim().isNotEmpty ?? false)
   211→            ? (json['name'] as String).trim()
   212→            : (json['tag_name'] as String? ?? 'Latest Plugin Release').trim(),
   213→        body: (json['body'] as String? ?? '').trim(),
   214→        publishedAt: DateTime.tryParse(json['published_at'] as String? ?? ''),
   215→        asset: PluginAsset(
   216→          name: (assetJson['name'] as String? ?? 'anymex_bridge').trim(),
   217→          downloadUrl:
   218→              (assetJson['browser_download_url'] as String? ?? '').trim(),
   219→          sizeBytes: (assetJson['size'] as num?)?.toInt() ?? 0,
   220→        ),
   221→      );
   222→    } catch (error, stackTrace) {
   223→      Logger.e(
   224→        'Error fetching runtime host release',
   225→        error: error,
   226→        stackTrace: stackTrace,
   227→      );
   228→      return null;
   229→    }
   230→  }

   232→  Future<List<PluginRelease>> fetchReleases() async {
   233→    try {
   234→      final response = await http.get(
   235→        Uri.parse('https://api.github.com/repos/RyanYuuki/AnymeXExtensionRuntimeBridge/releases'),
   236→        headers: const {'Accept': 'application/vnd.github+json'},
   237→      );
   238→      if (response.statusCode != 200) return [];
   239→      final List<dynamic> jsonList = jsonDecode(response.body) as List<dynamic>;
   240→      final List<PluginRelease> releases = [];
   241→      for (final json in jsonList) {
   242→        final assets = (json['assets'] as List<dynamic>? ?? [])
   243→            .whereType<Map<String, dynamic>>()
   244→            .toList();
   245→        final extension = Platform.isAndroid ? '.apk' : '.jar';
   246→        final assetJson = assets.firstWhereOrNull(
   247→          (asset) => (asset['name'] as String? ?? '').toLowerCase().endsWith(extension),
   248→        );
   249→        if (assetJson != null) {
   250→          releases.add(PluginRelease(
   251→            tagName: (json['tag_name'] as String? ?? '').trim(),
   252→            title: ((json['name'] as String?)?.trim().isNotEmpty ?? false)
   253→                ? (json['name'] as String).trim()
   254→                : (json['tag_name'] as String? ?? 'Plugin Release').trim(),
   255→            body: (json['body'] as String? ?? '').trim(),
   256→            publishedAt: DateTime.tryParse(json['published_at'] as String? ?? ''),
   257→            asset: PluginAsset(
   258→              name: (assetJson['name'] as String? ?? 'anymex_bridge').trim(),
   259→              downloadUrl: (assetJson['browser_download_url'] as String? ?? '').trim(),
   260→              sizeBytes: (assetJson['size'] as num?)?.toInt() ?? 0,
   261→            ),
   262→          ));
   263→        }
   264→      }
   265→      return releases;
   266→    } catch (_) {
   267→      return [];
   268→    }
   269→  }

   271→  bool isNewerVersion(String installed, String latest) {
   272→    final installedParts = _normalizeVersion(installed);
   273→    final latestParts = _normalizeVersion(latest);
   274→    final maxLength = installedParts.length > latestParts.length
   275→        ? installedParts.length
   276→        : latestParts.length;
   277→
   278→    for (var index = 0; index < maxLength; index++) {
   279→      final installedPart =
   280→          index < installedParts.length ? installedParts[index] : 0;
   281→      final latestPart = index < latestParts.length ? latestParts[index] : 0;
   282→
   283→      if (latestPart > installedPart) return true;
   284→      if (latestPart < installedPart) return false;
   285→    }
   286→
   287→    return false;
   288→  }
   289→
   290→  List<int> _normalizeVersion(String version) {
   291→    final cleaned = version.toLowerCase().replaceFirst(RegExp(r'^v'), '');
   292→    final stablePart = cleaned.split('-').first;
   293→    return stablePart
   294→        .split('.')
   295→        .map((part) => int.tryParse(part) ?? 0)
   296→        .toList();
   297→  }

   303→  Future<bool> syncLocalApk(String apkPath) async {
   304→    final isJar = apkPath.toLowerCase().endsWith('.jar');
   305→    final isApk = apkPath.toLowerCase().endsWith('.apk');
   306→
   307→    if (Platform.isAndroid && !isApk) {
   308→      errorSnackBar('Please select a valid APK file.');
   309→      return false;
   310→    }
   311→    if (!Platform.isAndroid && !isJar) {
   312→      errorSnackBar('Please select a valid JAR file.');
   313→      return false;
   314→    }
   315→
   316→    if (!await File(apkPath).exists()) {
   317→      errorSnackBar('Local file not found at: $apkPath');
   318→      return false;
   319→    }
   320→
   321→    try {
   322→      if (Platform.isAndroid) {
   323→        await AnymeXRuntimeBridge.useLocalApk(apkPath);
   324→      } else {
   325→        final paths = RuntimePaths();
   326→        final destPath = await paths.bridgePath;
   327→        final destFile = File(destPath);
   328→        if (await destFile.exists()) {
   329→          await destFile.delete();
   330→        }
   331→        await File(apkPath).copy(destPath);
   332→        
   333→        final toolsDir = await paths.toolsDir;
   334→        final metadataFile = File('${toolsDir.path}/metadata.json');
   335→        await metadataFile.writeAsString(jsonEncode({
   336→          'version': 'Local-${DateTime.now().millisecondsSinceEpoch}',
   337→          'title': 'Local Synced Jar',
   338→        }));
   339→        await AnymeXRuntimeBridge.loadMetadata();
   340→      }
   341→
   342→      final bridge = AnymeXRuntimeBridge.controller;
   343→      if (Platform.isAndroid && bridge.isReady.value) {
   344→        await Get.find<ExtensionManager>()
   345→            .onRuntimeBridgeInitialization(force: true);
   346→        successSnackBar('Plugin synced from local APK.');
   347→        return true;
   348→      } else if (!Platform.isAndroid) {
   349→        successSnackBar('Plugin jar copied. Please restart the app.');
   350→        return true;
   351→      }
   352→
   353→      errorSnackBar('Sync completed, but the plugin is not ready yet.');
   354→      return false;
   355→    } catch (e) {
   356→      errorSnackBar('Sync failed: $e');
   357→      return false;
   358→    }
   359→  }
   360→
   361→  Future<bool> forceSyncLocalApk() async {
   362→    const localPath = '/storage/emulated/0/AnymeX/anymex_runtime_host.apk';
   363→    if (!await File(localPath).exists()) {
   364→      errorSnackBar('Local APK not found at: $localPath');
   365→      return false;
   366→    }
   367→
   368→    return syncLocalApk(localPath);
   369→  }

   424→  void _checkBridgeReady() {
   425→    final bridge = AnymeXRuntimeBridge.controller;
   426→    if (bridge.isReady.value && !AnymeXRuntimeBridge.isPluginInstalled) {
   427→      if (_release != null) {
   428→        widget.manager.persistInstalledRelease(_release!);
   429→      }
   430→      _installFinished = true;
   431→      WidgetsBinding.instance.addPostFrameCallback((_) {
   432→        Get.find<ExtensionManager>().onRuntimeBridgeInitialization(force: true);
   433→      });
   434→    }
   435→  }

   437→  Future<void> _fetchReleaseDetails() async {
   438→    setState(() {
   439→      _isLoadingRelease = true;
   440→    });
   441→    try {
   442→      final fetched = await widget.manager.fetchLatestRelease();
   443→      if (mounted) {
   444→        setState(() {
   445→          _release = fetched;
   446→          _isLoadingRelease = false;
   447→        });
   448→        if (fetched != null) {
   449→          _checkBridgeReady();
   450→        } else {
   451→          errorSnackBar('Failed to fetch release details.');
   452→          Navigator.of(context).pop();
   453→        }
   454→      }
   455→    } catch (e) {
   456→      if (mounted) {
   457→        setState(() {
   458→          _isLoadingRelease = false;
   459→        });
   460→        errorSnackBar('Error: $e');
   461→        Navigator.of(context).pop();
   462→      }
   463→    }
   464→  }

   472→  Future<void> _downloadAndInstall() async {
   473→    if (_release == null) return;
   474→    final bridge = AnymeXRuntimeBridge.controller;
   475→    if (bridge.isDownloading.value) return;
   476→
   477→    try {
   478→      await AnymeXRuntimeBridge.setupRuntime(
   479→        force: widget.mode == _PluginSheetMode.update,
   480→      );
   481→
   482→      if (bridge.isReady.value) {
   483→        await Get.find<ExtensionManager>()
   484→            .onRuntimeBridgeInitialization(force: true);
   485→        widget.manager.persistInstalledRelease(_release!);
   486→        if (mounted) {
   487→          setState(() {
   488→            _installFinished = true;
   489→          });
   490→          successSnackBar(
   491→            widget.mode == _PluginSheetMode.install
   492→                ? 'Plugin installed successfully.'
   493→                : 'Plugin updated successfully.',
   494→          );
   495→        }
   496→      }
   497→    } catch (error) {
   498→      if (mounted) errorSnackBar('Plugin installation failed: $error');
   499→    }
   500→  }

   502→  String _formatBytes(int bytes) {
   503→    if (bytes <= 0) return 'Unknown';
   504→    const kb = 1024;
   505→    const mb = kb * 1024;
   506→    if (bytes < kb) return '$bytes B';
   507→    if (bytes < mb) return '${(bytes / kb).toStringAsFixed(1)} KB';
   508→    return '${(bytes / mb).toStringAsFixed(2)} MB';
   509→  }
