   132→    private fun loadAnymeXRuntimeHost(apkPath: String, settingsMap: Map<String, Any?>? = null): Boolean {
   133→        Log.i(TAG, "loadAnymeXRuntimeHost called with: $apkPath")
   134→        val ctx = context ?: run {
   135→            Log.e(TAG, "loadAnymeXRuntimeHost: context is null")
   136→            return false
   137→        }
   138→
   139→        return try {
   140→            runtimeBridge = null
   141→            bridgeClass = null
   142→
   143→            val originalApk = File(apkPath)
   144→            if (!originalApk.exists()) {
   145→                Log.e(TAG, "APK does not exist at path: $apkPath")
   146→                return false
   147→            }
   148→
   149→            val cacheApkName = "anymex_runtime_${originalApk.length()}_${originalApk.lastModified()}.apk"
   150→            val cacheApk = File(ctx.filesDir, cacheApkName)
   151→
   152→            if (!cacheApk.exists()) {
   153→                Log.i(TAG, "Creating new cached APK: $cacheApkName")
   154→                ctx.filesDir.listFiles()?.forEach { file ->
   155→                    if (file.name.startsWith("anymex_runtime_") && file.name.endsWith(".apk") && file.name != cacheApkName) {
   156→                        Log.d(TAG, "Deleting old cached APK: ${file.name}")
   157→                        file.delete()
   158→                    }
   159→                }
   160→
   161→                originalApk.inputStream().use { input ->
   162→                    FileOutputStream(cacheApk).use { output ->
   163→                        input.copyTo(output)
   164→                    }
   165→                }
   166→                cacheApk.setReadOnly()
   167→            } else {
   168→                Log.i(TAG, "Using existing cached APK: $cacheApkName")
   169→            }
   170→
   171→            ctx.cacheDir.listFiles()?.forEach { file ->
   172→                if (file.isDirectory && (file.name.startsWith("anymex_dex_") || file.name.startsWith("anymex_libs_"))) {
   173→                    file.deleteRecursively()
   174→                }
   175→            }
   176→
   177→            val optimizedDir = File(ctx.cacheDir, "anymex_dex_${System.currentTimeMillis()}")
   178→            optimizedDir.mkdirs()
   179→
   180→            val libsDir = File(ctx.cacheDir, "anymex_libs_${System.currentTimeMillis()}")
   181→            libsDir.mkdirs()
   182→
   183→            try {
   184→                java.util.zip.ZipFile(cacheApk).use { zip ->
   185→                    val abisList = android.os.Build.SUPPORTED_ABIS
   186→                    var selectedAbi: String? = null
   187→                    
   188→                    for (abi in abisList) {
   189→                        val prefix = "lib/$abi/"
   190→                        var found = false
   191→                        val entriesEnum = zip.entries()
   192→                        while (entriesEnum.hasMoreElements()) {
   193→                            val entry = entriesEnum.nextElement()
   194→                            if (entry.name.startsWith(prefix) && entry.name.endsWith(".so")) {
   195→                                found = true
   196→                                break
   197→                            }
   198→                        }
   199→                        if (found) {
   200→                            selectedAbi = abi
   201→                            break
   202→                        }
   203→                    }
   204→
   205→                    if (selectedAbi != null) {
   206→                        Log.i(TAG, "Extracting native libraries for ABI: $selectedAbi")
   207→                        val prefix = "lib/$selectedAbi/"
   208→                        val entriesEnum = zip.entries()
   209→                        while (entriesEnum.hasMoreElements()) {
   210→                            val entry = entriesEnum.nextElement()
   211→                            if (entry.name.startsWith(prefix) && entry.name.endsWith(".so")) {
   212→                                val libName = entry.name.substringAfterLast('/')
   213→                                val outFile = File(libsDir, libName)
   214→                                zip.getInputStream(entry).use { input ->
   215→                                    FileOutputStream(outFile).use { output ->
   216→                                        input.copyTo(output)
   217→                                    }
   218→                                }
   219→                                Log.d(TAG, "Extracted native library: $libName to ${outFile.absolutePath}")
   220→                            }
   221→                        }
   222→                    } else {
   223→                        Log.w(TAG, "No matching native libraries found in APK for supported ABIs: ${abisList.joinToString()}")
   224→                    }
   225→                }
   226→            } catch (e: Exception) {
   227→                Log.e(TAG, "Failed to extract native libraries: ${e.message}", e)
   228→            }
   229→
   230→            val loader = ChildFirstClassLoader(
   231→                cacheApk.absolutePath,
   232→                optimizedDir.absolutePath,
   233→                libsDir.absolutePath,
   234→                ctx.classLoader!!
   235→            )
   236→
   237→            bridgeClass = loader.loadClass("com.anymex.runtimehost.RuntimeBridge")
   238→            Log.d(TAG, "bridgeClass loaded: $bridgeClass")
   239→            runtimeBridge = bridgeClass!!.getField("INSTANCE").get(null)
   240→            
   241→            try {
   242→                val loggerClass = loader.loadClass("com.anymex.runtimehost.Logger")
   243→                val setLogCallbackMethod = loggerClass.getMethod("setLogCallback", Any::class.java, Method::class.java)
   244→                val ourLogMethod = AnymexExtensionRuntimeBridgePlugin::class.java.getMethod(
   245→                    "logFromHost",
   246→                    String::class.java,
   247→                    String::class.java,
   248→                    String::class.java
   249→                )
   250→                setLogCallbackMethod.invoke(null, this, ourLogMethod)
   251→                Log.i(TAG, "Logger callback registered successfully.")
   252→            } catch (e: Throwable) {
   253→                Log.e(TAG, "Failed to register Logger callback: ${e.message}")
   254→            }
   255→
   256→            Log.i(TAG, "AnymeX Runtime Bridge initialized successfully")
   257→            
   258→            try {
   259→                call("initialize", ctx, settingsMap)
   260→            } catch (e: Throwable) {
   261→                Log.e(TAG, "Failed to initialize RuntimeBridge: ${e.message}")
   262→                logToFlutter("ERROR", "BRIDGE_INIT", "Failed to initialize RuntimeBridge: ${e.message}\n${Log.getStackTraceString(e)}")
   263→            }
   264→
   265→            Log.i(TAG, "Runtime Host loaded successfully")
   266→            true
   267→        } catch (e: Throwable) {
   268→            Log.e(TAG, "Failed to load Runtime Host APK: ${e.message}")
   269→            logToFlutter("ERROR", "BRIDGE_LOAD", "Failed to load Runtime Host APK: ${e.message}\n${Log.getStackTraceString(e)}")
   270→            false
   271→        }
   272→    }

   274→    private fun handleAnymeX(call: MethodCall, result: MethodResult) {
   275→        when (call.method) {
   276→            "getAbi" -> {
   277→                val primaryAbi = android.os.Build.SUPPORTED_ABIS.firstOrNull() ?: "arm64-v8a"
   278→                result.success(primaryAbi)
   279→            }
   280→            "loadAnymeXRuntimeHost" -> {
   281→                val path = call.argument<String>("path")
   282→                val settingsMap = call.argument<Map<String, Any?>>("settings")
   283→                if (path.isNullOrBlank()) {
   284→                    result.error("INVALID_ARG", "path is required", null)
   285→                    return
   286→                }
   287→                scope.launch {
   288→                    try {
   289→                        val ok = loadAnymeXRuntimeHost(path, settingsMap)
   290→                        withContext(Dispatchers.Main) {
   291→                            if (ok) result.success(true)
   292→                            else result.error("LOAD_FAILED", "Failed to load runtime host APK", null)
   293→                        }
   294→                    } catch (e: Throwable) {
   295→                        sendError(result, "loadAnymeXRuntimeHost", e)
   296→                    }
   297→                }
   298→            }
   299→            "isLoaded" -> {
   300→                Log.d(TAG, "isLoaded check: runtimeBridge=$runtimeBridge, bridgeClass=$bridgeClass")
   301→                result.success(runtimeBridge != null)
   302→            }
   303→            "cancelRequest" -> {
   304→                val token = call.argument<String>("token")
   305→                if (token != null) {
   306→                    val ok = call("cancelRequest", token) as? Boolean ?: false
   307→                    result.success(ok)
   308→                } else {
   309→                    result.error("INVALID_ARG", "token is required", null)
   310→                }
   311→            }
   312→            "setCookies" -> {
   313→                val url = call.argument<String>("url")
   314→                val cookieString = call.argument<String>("cookieString")
   315→                if (url.isNullOrBlank() || cookieString == null) {
   316→                    result.error("INVALID_ARG", "url and cookieString are required", null)
   317→                    return
   318→                }
   319→                try {
   320→                    val mgr = android.webkit.CookieManager.getInstance()
   321→                    mgr.setAcceptCookie(true)
   322→                    cookieString.split(";").map { it.trim() }.filter { it.isNotEmpty() }.forEach { cookie ->
   323→                        mgr.setCookie(url, cookie)
   324→                    }
   325→                    mgr.flush()
   326→                    Log.d(TAG, "setCookies: applied ${cookieString.split(";").size} cookie(s) for $url")
   327→                    result.success(null)
   328→                } catch (e: Exception) {
   329→                    Log.e(TAG, "setCookies failed: ${e.message}")
   330→                    result.error("COOKIE_ERROR", e.message, null)
   331→                }
   332→            }
   333→            "setUserAgent" -> {
   334→                val url = call.argument<String>("url")
   335→                val userAgent = call.argument<String>("userAgent")
   336→                if (url.isNullOrBlank() || userAgent.isNullOrBlank()) {
   337→                    result.error("INVALID_ARG", "url and userAgent are required", null)
   338→                    return
   339→                }
   340→                try {
   341→                    val host = android.net.Uri.parse(url).host ?: url
   342→                    System.setProperty("anymex.ua.$host", userAgent)
   343→                    Log.d(TAG, "setUserAgent: stored UA for host=$host")
   344→                    result.success(null)
   345→                } catch (e: Exception) {
   346→                    Log.e(TAG, "setUserAgent failed: ${e.message}")
   347→                    result.error("UA_ERROR", e.message, null)
   348→                }
   349→            }
   350→            else -> result.notImplemented()
   351→
   352→        }
   353→    }

   485→    private fun handleCloudStream(call: MethodCall, result: MethodResult) {
   486→        if (!ensureLoaded(result)) return
   487→        val ctx = effectiveContext() ?: return result.error("NO_CTX", "No context", null)
   488→
   489→        when (call.method) {
   490→            "initialize" -> {
   491→                call("initialize", ctx)
   492→                result.success(null)
   493→                return
   494→            }
   495→            "getRegisteredProviders" -> {
   496→                result.success(call("csGetRegisteredProviders"))
   497→                return
   498→            }
   499→        }
   500→
   501→        scope.launch {
   502→            try {
   503→                val res: Any? = when (call.method) {
   504→                    "loadPlugin" -> {
   505→                        val path = call.argument<String>("path")
   506→                            ?: return@launch withContext(Dispatchers.Main) {
   507→                                result.error("INVALID_ARG", "path required", null)
   508→                            }
   509→                        call("csLoadPlugin", ctx, path)
   510→                    }
   511→                    "search" -> {
   512→                        call("csSearch", ctx,
   513→                            call.argument<String>("query") ?: "",
   514→                            call.argument<String>("apiName"),
   515→                            call.argument<Int>("page") ?: 1,
   516→                            call.argument<Map<String, Any?>>("parameters"))
   517→                    }
   518→                    "getDetail" -> {
   519→                        call("csGetDetail", ctx,
   520→                            call.argument<String>("apiName") ?: "",
   521→                            call.argument<String>("url") ?: "",
   522→                            call.argument<Map<String, Any?>>("parameters"))
   523→                    }
   524→                    "getVideoList" -> {
   525→                        call("csGetVideoList", ctx,
   526→                            call.argument<String>("apiName") ?: "",
   527→                            call.argument<String>("url") ?: "",
   528→                            call.argument<Map<String, Any?>>("parameters"))
   529→                    }
   530→                    "deletePlugin" -> {
   531→                        val internalName = call.argument<String>("internalName")
   532→                            ?: return@launch withContext(Dispatchers.Main) {
   533→                                result.error("INVALID_ARG", "internalName required", null)
   534→                            }
   535→                        call("csUnloadPlugin", internalName)
   536→                    }
   537→                    "getExtensionSettings" -> {
   538→                        call("csGetExtensionSettings", ctx,
   539→                            call.argument<String>("pluginName") ?: "")
   540→                    }
   541→                    "setExtensionSettings" -> {
   542→                        call("csSetExtensionSettings", ctx,
   543→                            call.argument<String>("pluginName") ?: "",
   544→                            call.argument<String>("key") ?: "",
   545→                            call.argument<Any>("value"))
   546→                    }
   547→                    "openSettings" -> {
   548→                        val pluginName = call.argument<String>("pluginName") ?: ""

   622→    private fun sendError(result: MethodResult, methodName: String, e: Throwable) {
   623→        val realError = (e as? java.lang.reflect.InvocationTargetException)?.targetException ?: e
   624→        
   625→        val stackTrace = Log.getStackTraceString(e) 
   626→        val errorMessage = realError.message ?: realError.toString()
   627→        val detailedError = "Method: $methodName\nError: $errorMessage\n$stackTrace"
   628→        
   629→        Log.e(TAG, detailedError)
   630→        logToFlutter("ERROR", "BRIDGE", detailedError)
   631→        
   632→        scope.launch(Dispatchers.Main) {
   633→            result.error("BRIDGE_ERROR", errorMessage, detailedError)
   634→        }
   635→    }

   637→    private fun call(methodName: String, vararg args: Any?): Any? {
   638→        val bridge = runtimeBridge ?: throw IllegalStateException("Runtime Host not loaded")
   639→        val cls = bridgeClass ?: throw IllegalStateException("Runtime Host class not loaded")
   640→
   641→        val method = cls.methods.filter { it.name == methodName }
   642→            .firstOrNull { it.parameterTypes.size == args.size }
   643→            ?: cls.methods.firstOrNull { it.name == methodName }
   644→            ?: throw NoSuchMethodException("No method '$methodName' in RuntimeBridge")
   645→
   646→        val effectiveArgs = if (method.parameterTypes.size != args.size) {
   647→            logToFlutter("WARNING", "BRIDGE", "Argument count mismatch for $methodName. Expected ${method.parameterTypes.size}, got ${args.size}. Adjusting.")
   648→            if (args.size > method.parameterTypes.size) {
   649→                args.take(method.parameterTypes.size).toTypedArray()
   650→            } else {
   651→                val padded = args.toMutableList()
   652→                while (padded.size < method.parameterTypes.size) padded.add(null)
   653→                padded.toTypedArray()
   654→            }
   655→        } else {
   656→            args
   657→        }
   658→
   659→        logToFlutter("INFO", "BRIDGE", "Calling Method: RuntimeBridge.$methodName")
   660→        val result = method.invoke(bridge, *effectiveArgs)
   661→        logToFlutter("INFO", "BRIDGE", "Method '$methodName' completed successfully")
   662→
   663→        return result
   664→    }

   746→    private fun installSourceInternal(context: Context, apkPath: String, isAnime: Boolean): Boolean {
   747→        return try {
   748→            val pm = context.packageManager
   749→            val packageInfo = pm.getPackageArchiveInfo(apkPath, 0)
   750→                ?: throw IllegalArgumentException("Invalid APK file at $apkPath")
   751→            val packageName = packageInfo.packageName
   752→
   753→            val dirName = if (isAnime) "exts" else "exts_manga"
   754→            val privateDir = File(context.filesDir, dirName)
   755→            if (!privateDir.exists()) {
   756→                privateDir.mkdirs()
   757→            }
   758→
   759→            val srcFile = File(apkPath)
   760→            val dstFile = File(privateDir, "$packageName.apk")
   761→            val tmpFile = File(privateDir, "$packageName.apk.tmp")
   762→
   763→            srcFile.inputStream().use { input ->
   764→                FileOutputStream(tmpFile).use { output ->
   765→                    input.copyTo(output)
   766→                }
   767→            }
   768→
   769→            if (dstFile.exists()) dstFile.delete()
   770→
   771→            if (tmpFile.renameTo(dstFile)) {
   772→                try {
   773→                    dstFile.setReadOnly()
   774→                } catch (_: Exception) {}
   775→                Log.i(TAG, "Successfully installed internal extension: $packageName to ${dstFile.absolutePath}")
   776→                true
   777→            } else {
   778→                tmpFile.delete()
   779→                false
   780→            }
   781→        } catch (e: Exception) {
   782→            Log.e(TAG, "Failed to install source internally: ${e.message}", e)
   783→            false
   784→        }
   785→    }

   807→    private class ChildFirstClassLoader(
   808→        dexPath: String,
   809→        optimizedDirectory: String?,
   810→        librarySearchPath: String?,
   811→        parent: ClassLoader
   812→    ) : DexClassLoader(dexPath, optimizedDirectory, librarySearchPath, parent) {
   813→
   814→        private val systemClassLoader: ClassLoader? = getSystemClassLoader()
   815→
   816→        private fun shouldDelegateToParent(name: String?): Boolean {
   817→            if (name == null) return false
   818→            return name.startsWith("androidx.")
   819→        }
   820→
   821→        override fun loadClass(name: String?, resolve: Boolean): Class<*> {
   822→            var c = findLoadedClass(name)
   823→
   824→            if (c == null && systemClassLoader != null) {
   825→                try {
   826→                    c = systemClassLoader.loadClass(name)
   827→                } catch (_: ClassNotFoundException) {}
   828→            }
   829→
   830→            if (c == null && shouldDelegateToParent(name)) {
   831→                try {
   832→                    c = parent.loadClass(name)
   833→                } catch (_: ClassNotFoundException) {}
   834→            }
   835→
   836→            if (c == null) {
   837→                try {
   838→                    c = findClass(name)
   839→                } catch (_: ClassNotFoundException) {
   840→                    c = super.loadClass(name, resolve)
   841→                }
   842→            }
   843→
   844→            if (resolve) {
   845→                resolveClass(c)
   846→            }
   847→
   848→            return c
   849→        }
   850→    }
