   432→   432→    @android.annotation.SuppressLint("RestrictedApi")
   433→   433→    fun aniyomiGetPreference(
   434→   434→        context: Context,
   435→   435→        sourceId: String,
   436→   436→        isAnime: Boolean,
   437→   437→    ): List<Map<String, Any?>> = runBlocking {
   438→   438→        sourcePreferences.remove(sourceId)
   439→   439→        
   440→   440→        val prefManager = androidx.preference.PreferenceManager(context)
   441→   441→        prefManager.sharedPreferencesName = "source_$sourceId"
   442→   442→        
   443→   443→        val screen = prefManager.createPreferenceScreen(context)
   444→   444→        
   445→   445→        try {
   446→   446→            media(context, sourceId, isAnime).setupPreferenceScreen(screen)
   447→   447→        } catch (e: com.anymex.runtimehost.aniyomi.NoPreferenceScreenException) {
   448→   448→            return@runBlocking emptyList<Map<String, Any?>>()
   449→   449→        }
   450→   450→
   451→   451→        val list = mutableListOf<Map<String, Any?>>()
   452→   452→        val store = sourcePreferences.getOrPut(sourceId) { mutableMapOf() }
   453→   453→
   454→   454→        fun walk(group: androidx.preference.PreferenceGroup) {
   455→   455→            for (i in 0 until group.preferenceCount) {
   456→   456→                val p = group.getPreference(i)
   457→   457→                store[p.key] = PrefHandlers(p, p.onPreferenceClickListener, p.onPreferenceChangeListener)
   458→   458→
   459→   459→                val map = mutableMapOf(
   460→   460→                    "key" to p.key,
   461→   461→                    "title" to p.title?.toString(),
   462→   462→                    "summary" to p.summary?.toString(),
   463→   463→                    "enabled" to p.isEnabled,
   464→   464→                    "type" to when (p) {

   875→   875→    private fun parseEpisodeInfoFromName(name: String): Pair<Float, Int?> {
   876→   876→        val seasonRegex = Regex("""(?i)\b(?:s|season)\s*(\d+)""")
   877→   877→        val seasonMatch = seasonRegex.find(name)
   878→   878→        val season = seasonMatch?.groupValues?.get(1)?.toIntOrNull()
   879→   879→
   880→   880→        val cleanName = if (seasonMatch != null) name.replace(seasonMatch.value, "") else name
   881→   881→
   882→   882→        val prefixRegex = Regex("""(?i)(?:chapter|ch\.|ch|ep\.|ep|episode|e)\s*(\d+(\.\d+)?)""")
   883→   883→        val prefixMatch = prefixRegex.find(cleanName)
   884→   884→        if (prefixMatch != null) {
   885→   885→            val epVal = prefixMatch.groupValues[1].toFloatOrNull() ?: -1f
   886→   886→            return Pair(epVal, season)
   887→   887→        }
   888→   888→
   889→   889→        val fallbackRegex = Regex("""(\d+(\.\d+)?)""")
   890→   890→        val epVal = fallbackRegex.find(cleanName)?.groupValues?.get(1)?.toFloatOrNull() ?: -1f
   891→   891→        return Pair(epVal, season)
   892→   892→    }
