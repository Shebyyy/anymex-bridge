   179→  Future<PluginRelease?> fetchLatestRelease() async {
   180→    try {
   181→      final response = await http.get(
   182→        Uri.parse(_latestReleaseUrl),
   183→        headers: const {'Accept': 'application/vnd.github+json'},
   184→      );
   185→
   186→      if (response.statusCode != 200) {
   187→        Logger.e(
   188→          'Failed to fetch runtime host release: ${response.statusCode}',
   189→        );
   190→        return null;
   191→      }
   192→
   193→      final Map<String, dynamic> json =
   194→          jsonDecode(response.body) as Map<String, dynamic>;
   195→      final assets = (json['assets'] as List<dynamic>? ?? [])
   196→          .whereType<Map<String, dynamic>>()
   197→          .toList();
   198→
   199→      final extension = Platform.isAndroid ? '.apk' : '.jar';
   200→      final assetJson = assets.firstWhereOrNull(
   201→        (asset) =>
   202→            (asset['name'] as String? ?? '').toLowerCase().endsWith(extension),
   203→      );
   204→
   205→      if (assetJson == null) {
   206→        Logger.e('Runtime host release has no compatible asset.');
   207→        return null;
   208→      }
   209→
   210→      return PluginRelease(
   211→        tagName: (json['tag_name'] as String? ?? '').trim(),
   212→        title: ((json['name'] as String?)?.trim().isNotEmpty ?? false)
   213→            ? (json['name'] as String).trim()
   214→            : (json['tag_name'] as String? ?? 'Latest Plugin Release').trim(),
   215→        body: (json['body'] as String? ?? '').trim(),
   216→        publishedAt: DateTime.tryParse(json['published_at'] as String? ?? ''),
   217→        asset: PluginAsset(
   218→          name: (assetJson['name'] as String? ?? 'anymex_bridge').trim(),
   219→          downloadUrl:
   220→              (assetJson['browser_download_url'] as String? ?? '').trim(),
   221→          sizeBytes: (assetJson['size'] as num?)?.toInt() ?? 0,
   222→        ),
   223→      );
   224→    } catch (error, stackTrace) {
   225→      Logger.e(
   226→        'Error fetching runtime host release',
   227→        error: error,
   228→        stackTrace: stackTrace,
   229→      );
   230→      return null;
   231→    }
   232→  }

   234→  Future<List<PluginRelease>> fetchReleases() async {
   235→    try {
   236→      final response = await http.get(
   237→        Uri.parse('https://api.github.com/repos/RyanYuuki/AnymeXExtensionRuntimeBridge/releases'),
   238→        headers: const {'Accept': 'application/vnd.github+json'},
   239→      );
   240→      if (response.statusCode != 200) return [];
   241→      final List<dynamic> jsonList = jsonDecode(response.body) as List<dynamic>;
   242→      final List<PluginRelease> releases = [];
   243→      for (final json in jsonList) {
   244→        final assets = (json['assets'] as List<dynamic>? ?? [])
   245→            .whereType<Map<String, dynamic>>()
   246→            .toList();
   247→        final extension = Platform.isAndroid ? '.apk' : '.jar';
   248→        final assetJson = assets.firstWhereOrNull(
   249→          (asset) => (asset['name'] as String? ?? '').toLowerCase().endsWith(extension),
   250→        );
   251→        if (assetJson != null) {
   252→          releases.add(PluginRelease(
   253→            tagName: (json['tag_name'] as String? ?? '').trim(),
   254→            title: ((json['name'] as String?)?.trim().isNotEmpty ?? false)
   255→                ? (json['name'] as String).trim()
   256→                : (json['tag_name'] as String? ?? 'Plugin Release').trim(),
   257→            body: (json['body'] as String? ?? '').trim(),
   258→            publishedAt: DateTime.tryParse(json['published_at'] as String? ?? ''),
   259→            asset: PluginAsset(
   260→              name: (assetJson['name'] as String? ?? 'anymex_bridge').trim(),
   261→              downloadUrl: (assetJson['browser_download_url'] as String? ?? '').trim(),
   262→              sizeBytes: (assetJson['size'] as num?)?.toInt() ?? 0,
   263→            ),
   264→          ));
   265→        }
   266→      }
   267→      return releases;
   268→    } catch (_) {
   269→      return [];
   270→    }
   271→  }

   273→  bool isNewerVersion(String installed, String latest) {
   274→    final installedParts = _normalizeVersion(installed);
   275→    final latestParts = _normalizeVersion(latest);
   276→    final maxLength = installedParts.length > latestParts.length
   277→        ? installedParts.length
   278→        : latestParts.length;
   279→
   280→    for (var index = 0; index < maxLength; index++) {
   281→      final installedPart =
   282→          index < installedParts.length ? installedParts[index] : 0;
   283→      final latestPart = index < latestParts.length ? latestParts[index] : 0;
   284→
   285→      if (latestPart > installedPart) return true;
   286→      if (latestPart < installedPart) return false;
   287→    }
   288→
   289→    return false;
   290→  }
   291→
   292→  List<int> _normalizeVersion(String version) {
   293→    final cleaned = version.toLowerCase().replaceFirst(RegExp(r'^v'), '');
   294→    final stablePart = cleaned.split('-').first;
   295→    return stablePart
   296→        .split('.')
   297→        .map((part) => int.tryParse(part) ?? 0)
   298→        .toList();
   299→  }

   305→  Future<bool> syncLocalApk(String apkPath) async {
   306→    final isJar = apkPath.toLowerCase().endsWith('.jar');
   307→    final isApk = apkPath.toLowerCase().endsWith('.apk');
   308→
   309→    if (Platform.isAndroid && !isApk) {
   310→      errorSnackBar('Please select a valid APK file.');
   311→      return false;
   312→    }
   313→    if (!Platform.isAndroid && !isJar) {
   314→      errorSnackBar('Please select a valid JAR file.');
   315→      return false;
   316→    }
   317→
   318→    if (!await File(apkPath).exists()) {
   319→      errorSnackBar('Local file not found at: $apkPath');
   320→      return false;
   321→    }
   322→
   323→    try {
   324→      if (Platform.isAndroid) {
   325→        await AnymeXRuntimeBridge.useLocalApk(apkPath);
   326→      } else {
   327→        final paths = RuntimePaths();
   328→        final destPath = await paths.bridgePath;
   329→        final destFile = File(destPath);
   330→        if (await destFile.exists()) {
   331→          await destFile.delete();
   332→        }
   333→        await File(apkPath).copy(destPath);
   334→        
   335→        final toolsDir = await paths.toolsDir;
   336→        final metadataFile = File('${toolsDir.path}/metadata.json');
   337→        await metadataFile.writeAsString(jsonEncode({
   338→          'version': 'Local-${DateTime.now().millisecondsSinceEpoch}',
   339→          'title': 'Local Synced Jar',
   340→        }));
   341→        await AnymeXRuntimeBridge.loadMetadata();
   342→      }
   343→
   344→      final bridge = AnymeXRuntimeBridge.controller;
   345→      if (Platform.isAndroid && bridge.isReady.value) {
   346→        await Get.find<ExtensionManager>()
   347→            .onRuntimeBridgeInitialization(force: true);
   348→        successSnackBar('Plugin synced from local APK.');
   349→        return true;
   350→      } else if (!Platform.isAndroid) {
   351→        successSnackBar('Plugin jar copied. Please restart the app.');
   352→        return true;
   353→      }
   354→
   355→      errorSnackBar('Sync completed, but the plugin is not ready yet.');
   356→      return false;
   357→    } catch (e) {
   358→      errorSnackBar('Sync failed: $e');
   359→      return false;
   360→    }
   361→  }
   362→
   363→  Future<bool> forceSyncLocalApk() async {
   364→    const localPath = '/storage/emulated/0/AnymeX/anymex_runtime_host.apk';
   365→    if (!await File(localPath).exists()) {
   366→      errorSnackBar('Local APK not found at: $localPath');
   367→      return false;
   368→    }
   369→
   370→    return syncLocalApk(localPath);
   371→  }

   426→  void _checkBridgeReady() {
   427→    final bridge = AnymeXRuntimeBridge.controller;
   428→    if (bridge.isReady.value && !AnymeXRuntimeBridge.isPluginInstalled) {
   429→      if (_release != null) {
   430→        widget.manager.persistInstalledRelease(_release!);
   431→      }
   432→      _installFinished = true;
   433→      WidgetsBinding.instance.addPostFrameCallback((_) {
   434→        Get.find<ExtensionManager>().onRuntimeBridgeInitialization(force: true);
   435→      });
   436→    }
   437→  }

   439→  Future<void> _fetchReleaseDetails() async {
   440→    setState(() {
   441→      _isLoadingRelease = true;
   442→    });
   443→    try {
   444→      final fetched = await widget.manager.fetchLatestRelease();
   445→      if (mounted) {
   446→        setState(() {
   447→          _release = fetched;
   448→          _isLoadingRelease = false;
   449→        });
   450→        if (fetched != null) {
   451→          _checkBridgeReady();
   452→        } else {
   453→          errorSnackBar('Failed to fetch release details.');
   454→          Navigator.of(context).pop();
   455→        }
   456→      }
   457→    } catch (e) {
   458→      if (mounted) {
   459→        setState(() {
   460→          _isLoadingRelease = false;
   461→        });
   462→        errorSnackBar('Error: $e');
   463→        Navigator.of(context).pop();
   464→      }
   465→    }
   466→  }

   474→  Future<void> _downloadAndInstall() async {
   475→    if (_release == null) return;
   476→    final bridge = AnymeXRuntimeBridge.controller;
   477→    if (bridge.isDownloading.value) return;
   478→
   479→    try {
   480→      await AnymeXRuntimeBridge.setupRuntime(
   481→        force: widget.mode == _PluginSheetMode.update,
   482→      );
   483→
   484→      if (bridge.isReady.value) {
   485→        await Get.find<ExtensionManager>()
   486→            .onRuntimeBridgeInitialization(force: true);
   487→        widget.manager.persistInstalledRelease(_release!);
   488→        if (mounted) {
   489→          setState(() {
   490→            _installFinished = true;
   491→          });
   492→          successSnackBar(
   493→            widget.mode == _PluginSheetMode.install
   494→                ? 'Plugin installed successfully.'
   495→                : 'Plugin updated successfully.',
   496→          );
   497→        }
   498→      }
   499→    } catch (error) {
   500→      if (mounted) errorSnackBar('Plugin installation failed: $error');
   501→    }
   502→  }

   504→  String _formatBytes(int bytes) {
   505→    if (bytes <= 0) return 'Unknown';
   506→    const kb = 1024;
   507→    const mb = kb * 1024;
   508→    if (bytes < kb) return '$bytes B';
   509→    if (bytes < mb) return '${(bytes / kb).toStringAsFixed(1)} KB';
   510→    return '${(bytes / mb).toStringAsFixed(2)} MB';
   511→  }
