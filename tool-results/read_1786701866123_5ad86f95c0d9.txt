    65→    65→  void _showUpdatePopup() async {
    66→    66→    final release = await _pluginManager.fetchLatestRelease();

    68→    68→    if (release == null) {
    69→    69→      errorSnackBar('Failed to check for updates.');
    70→    70→      return;
    71→    71→    }
    72→    72→    final currentVersion = _installedVersion;
    73→    73→    if (currentVersion.isEmpty) {
    74→    74→      _showInstallPopup();
    75→    75→      return;
    76→    76→    }
    77→    77→    if (_pluginManager.isNewerVersion(currentVersion, release.tagName)) {
    78→    78→      final oldVersion = _installedVersion;
    79→    79→      await _pluginManager.showUpdateSheet(
    80→    80→        context,
    81→    81→        release: release,
    82→    82→        installedVersion: currentVersion,
    83→    83→      );

    85→    85→        if (_installedVersion != oldVersion) {
    86→    86→          setState(() {
    87→    87→            _needsRestart = true;
    88→    88→          });
    89→    89→        } else {
    90→    90→          setState(() {});
    91→    91→        }
    92→    92→      }
    93→    93→    } else {
    94→    94→      successSnackBar('Plugin is already up to date.');
    95→    95→    }
    96→    96→  }

    98→    98→  void _checkForUpdates() async {
    99→    99→    setState(() => _isCheckingUpdate = true);
   100→   100→    try {
   101→   101→      final release = await _pluginManager.fetchLatestRelease();
   102→   102→      if (!mounted) return;
   103→   103→      if (release == null) {
   104→   104→        errorSnackBar('Failed to check for updates.');
   105→   105→        return;
   106→   106→      }
   107→   107→      final currentVersion = _installedVersion;
   108→   108→      if (currentVersion.isEmpty) {
   109→   109→        _showInstallPopup();
   110→   110→        return;
   111→   111→      }
   112→   112→      if (_pluginManager.isNewerVersion(currentVersion, release.tagName)) {
   113→   113→        _showUpdatePopup();
   114→   114→      } else {
   115→   115→        successSnackBar('Plugin is already up to date.');
   116→   116→      }
   117→   117→    } finally {
   118→   118→      if (mounted) setState(() => _isCheckingUpdate = false);
   119→   119→    }
   120→   120→  }

   122→   122→  void _syncLocalApk() async {
   123→   123→    if (_isSyncingLocalApk) return;
   124→   124→
   125→   125→    final result = await FilePicker.platform.pickFiles(
   126→   126→      type: FileType.custom,
   127→   127→      allowedExtensions: Platform.isAndroid ? const ['apk'] : const ['jar'],
   128→   128→      allowMultiple: false,
   129→   129→      withData: false,
   130→   130→    );
   131→   131→
   132→   132→    final apkPath = result?.files.single.path;
   133→   133→    if (apkPath == null || apkPath.isEmpty) return;
   134→   134→    if (!mounted) return;
   135→   135→
   136→   136→    setState(() => _isSyncingLocalApk = true);
   137→   137→    try {
   138→   138→      final oldVersion = _installedVersion;
   139→   139→      final synced = await _pluginManager.syncLocalApk(apkPath);
   140→   140→      if (mounted && synced) {
   141→   141→        _checkStorageStatus();
   142→   142→        if (_installedVersion != oldVersion) {
   143→   143→          setState(() {
   144→   144→            _needsRestart = true;
   145→   145→          });
   146→   146→        } else {
   147→   147→          setState(() {});
   148→   148→        }
   149→   149→      }
   150→   150→    } finally {
   151→   151→      if (mounted) setState(() => _isSyncingLocalApk = false);
   152→   152→    }
   153→   153→  }

   294→   294→  void _performRollback(PluginRelease release) async {
   295→   295→    final bridge = AnymeXRuntimeBridge.controller;
   296→   296→    if (bridge.isDownloading.value) return;
   297→   297→    try {
   298→   298→      await AnymeXRuntimeBridge.setupRuntime(
   299→   299→        customDownloadUrl: release.asset.downloadUrl,
   300→   300→        force: true,
   301→   301→      );
   302→   302→      if (bridge.isReady.value) {
   303→   303→        await Get.find<ExtensionManager>()
   304→   304→            .onRuntimeBridgeInitialization(force: true);
   305→   305→        _pluginManager.persistInstalledRelease(release);
   306→   306→        _checkStorageStatus();
   307→   307→        if (mounted) {
   308→   308→          setState(() {
   309→   309→            _needsRestart = true;
   310→   310→          });
   311→   311→          successSnackBar(
   312→   312→              'Rollback to ${release.tagName} successful. Restart app to apply.');
   313→   313→        }
   314→   314→      }
   315→   315→    } catch (error) {
   316→   316→      if (mounted) errorSnackBar('Rollback failed: $error');
   317→   317→    }
