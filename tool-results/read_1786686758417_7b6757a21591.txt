    49→  List<Map<String, dynamic>> get _displayManagers {
    50→    final List<Map<String, dynamic>> list = [];
    51→    final activeManagers = em.managers;
    52→
    53→    for (final m in activeManagers) {
    54→      if (Platform.isIOS && m.requiresPlugin) continue;
    55→      list.add({
    56→        'name': m.name,
    57→        'manager': m,
    58→        'isMock': false,
    59→      });
    60→    }
    61→
    62→    if (!Platform.isIOS) {
    63→      final hasAniyomi = activeManagers.any((m) => m.name.toLowerCase().contains('aniyomi'));
    64→      final hasCloudStream = activeManagers.any((m) => m.name.toLowerCase().contains('cloudstream'));
    65→
    66→      if (!hasAniyomi) {
    67→        list.add({
    68→          'name': 'Aniyomi',
    69→          'manager': null,
    70→          'isMock': true,
    71→        });
    72→      }
    73→      if (!hasCloudStream) {
    74→        list.add({
    75→          'name': 'CloudStream',
    76→          'manager': null,
    77→          'isMock': true,
    78→        });
    79→      }
    80→    }
    81→
    82→    list.sort((a, b) {
    83→      final managerA = a['manager'] as Extension?;
    84→      final managerB = b['manager'] as Extension?;
    85→
    86→      final aRequires = managerA?.requiresPlugin ?? _isMockPluginRequired(a['name'] as String);
    87→      final bRequires = managerB?.requiresPlugin ?? _isMockPluginRequired(b['name'] as String);
    88→
    89→      if (aRequires != bRequires) {
    90→        return aRequires ? 1 : -1;
    91→      }
    92→
    93→      final aName = (a['name'] as String).toLowerCase();
    94→      final bName = (b['name'] as String).toLowerCase();
    95→      if (aName.contains('mangayomi')) return -1;
    96→      if (bName.contains('mangayomi')) return 1;
    97→
    98→      return aName.compareTo(bName);
    99→    });
   100→
   101→    return list;
   102→  }
   103→
   104→  bool _isMockPluginRequired(String name) {
   105→    final lower = name.toLowerCase();
   106→    return lower.contains('aniyomi') || lower.contains('cloudstream');
   107→  }

   110→  Future<void> _addRepos(ItemType type, List<String> urls) async {
   111→    final currentManagerData = _displayManagers[_managerIndex];
   112→    if (currentManagerData['isMock'] == true) return;
   113→    final activeManager = currentManagerData['manager'] as Extension;
   114→    await em.addRepos(urls, type, activeManager.id);
   115→    widget.onSave?.call();
   116→    if (mounted) setState(() {});
   117→  }
   118→
   119→  Future<void> _removeRepo(Repo repo, ItemType type) async {
   120→    final key = '${repo.url}_${type.name}';
   121→    setState(() => _deleting[key] = true);
   122→    try {
   123→      await em.removeRepo(repo, type);
   124→      widget.onSave?.call();
   125→    } catch (_) {
   126→      snackBar('Failed to remove repo');
   127→    } finally {
   128→      if (mounted) setState(() => _deleting.remove(key));
   129→    }
   130→  }

   460→        if (AnymeXRuntimeBridge.controller.error.value.isNotEmpty) {
   461→          return Center(
   462→            child: Padding(

   478→                      color: context.colors.onSurface,

   496→                      backgroundColor: context.colors.primary,

   541→        duration: const Duration(milliseconds: 200),

   543→            ? _buildEmpty(key: ValueKey('empty_$_tab'))
   544→            : _buildRepoList(repos, key: ValueKey('list_$_tab')),
   545→      );
   546→    });
   547→  }

   746→  Widget? _buildFab() {
   747→    if (_managerIndex >= _displayManagers.length) return null;
   748→    final selectedManagerData = _displayManagers[_managerIndex];
   749→    if (selectedManagerData['isMock'] == true) return null;
   750→
   751→    final manager = selectedManagerData['manager'] as Extension;
   752→    final supported = _tab == ItemType.anime
   753→        ? manager.supportsAnime
   754→        : _tab == ItemType.manga
   755→            ? manager.supportsManga
   756→            : manager.supportsNovel;
   757→    if (!supported) return null;
   758→    final colors = context.colors;
   759→    return FloatingActionButton.extended(
   760→      onPressed: _openAddDialog,
   761→      icon: const Icon(Icons.add, size: 20),
   762→      label: const Text('Add Repo',
   763→          style: TextStyle(fontWeight: FontWeight.w600, fontSize: 14)),
   764→      backgroundColor: colors.primary,
   765→      foregroundColor: colors.onPrimary,
   766→      elevation: 2,
   767→      shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(16)),
   768→    );
   769→  }

   784→  String _path(String url) {
   785→    try {
   786→      final p = Uri.parse(url).path;
   787→      return p.isEmpty ? url : p;
   788→    } catch (_) {
   789→      return url;
   790→    }
   791→  }
   792→
   793→  String? _host(String url) {
   794→    try {
   795→      final h = Uri.parse(url).host;
   796→      return h.isEmpty ? null : h;
   797→    } catch (_) {
   798→      return null;
   799→    }
   800→  }
