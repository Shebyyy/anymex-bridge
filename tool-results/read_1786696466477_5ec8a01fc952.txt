   292→   491→   491→function populateSourceDropdown(){
   293→   492→   492→  const sel=document.getElementById('invokeSource');

   298→   497→   497→  // Group by type, then by base name

   300→   499→   499→  _sources.forEach(s=>{
   301→   500→   500→    const t=s.type||'other';
   302→   501→   501→    if(!byType[t])byType[t]=[];

   310→   509→   509→    sources.forEach(s=>{

   314→   513→   513→    });
   315→   514→   514→    // Sort base names
   316→   515→   515→    [...groups.keys()].sort().forEach(bn=>{
   317→   516→   516→      const items=groups.get(bn);
   318→   517→   517→      if(items.length===1){
   319→   518→   518→        const s=items[0];
   320→   519→   519→        html+=`<option value="${s.id}" data-type="${s.type}" data-pkg="${esc(s.pkg)}">${esc(s.name)}</option>`;
   321→   520→   520→      }else{
   322→   521→   521→        // Multiple languages — use optgroup
   323→   522→   522→        html+=`<optgroup label="${esc(bn)}">`;
   324→   523→   523→        items.sort((a,b)=>(a.lang||'').localeCompare(b.lang||'')).forEach(s=>{
   325→   524→   524→          const label=s.lang?s.lang:s.name.replace(bn,'').replace(/^[\s\-–—(]+|[\s\-–—)]+$/g,'')||'?';
   326→   525→   525→          html+=`<option value="${s.id}" data-type="${s.type}" data-pkg="${esc(s.pkg)}">${esc(s.name)}</option>`;
   327→   526→   526→        });
   328→   527→   527→        html+='</optgroup>';
   329→   528→   528→      }
   330→   529→   529→    });
   331→   530→   530→    html+='</optgroup>';
   332→   531→   531→  }
   333→   532→   532→  sel.innerHTML=html;
   334→   533→   533→}
