    73→    @JvmOverloads
    74→    fun initialize(context: Context, settingsMap: Map<String, Any?>? = null) {
    75→        if (initialized) return
    76→
    77→        Log.i(TAG, "Initializing Runtime Host - Version: 1.0.6")
    78→
    79→        try {
    80→            Injekt.addSingletonFactory<android.app.Application> {
    81→                context.applicationContext as android.app.Application
    82→            }
    83→            Injekt.addSingletonFactory { NetworkHelper(context.applicationContext) }
    84→            Injekt.addSingletonFactory { Injekt.get<NetworkHelper>().client }
    85→            Injekt.addSingletonFactory {
    86→                Json {
    87→                    ignoreUnknownKeys = true
    88→                    explicitNulls = false
    89→                }
    90→            }
    91→
    92→            DataStore.init(context.applicationContext)
    93→            AcraApplication.context = context.applicationContext
    94→            CloudStreamApp.context = context.applicationContext
    95→
    96→            extensionManager = AniyomiExtensionManager(context.applicationContext)
    97→            Injekt.addSingletonFactory<AniyomiExtensionManager> { extensionManager!! }
    98→
    99→            val aniyomiExtensionsPath = settingsMap?.get("aniyomiExtensionsPath") as? String
   100→            if (aniyomiExtensionsPath != null) {
   101→                Log.i(TAG, "Proactive Loading: Starting Aniyomi extension scan at $aniyomiExtensionsPath")
   102→                scope.launch {
   103→                    getInstalledAnimeExtensions(context.applicationContext, aniyomiExtensionsPath)
   104→                    getInstalledMangaExtensions(context.applicationContext)
   105→                    Log.i(TAG, "Proactive Loading: Aniyomi background scan completed")
   106→                }
   107→            }
   108→        } catch (e: Throwable) {
   109→            Log.e(TAG, "Init Step 2 Failed (Skipping Critical Setup): ${e.message}")
   110→        }
   111→
   112→        Log.i(TAG, "Runtime Host Initialized successfully")
   113→        try {
   114→            val cacheDir = File(context.cacheDir, "manga_pages_cache")
   115→            if (cacheDir.exists()) {
   116→                cacheDir.deleteRecursively()
   117→            }
   118→        } catch (_: Exception) {}
   119→        try {
   120→            MangaImageProxy.start()
   121→        } catch (e: Exception) {
   122→            Log.e(TAG, "Failed to start MangaImageProxy: ${e.message}")
   123→        }
   124→        initialized = true
   125→    }

   127→    fun cancelRequest(token: String): Boolean {
   128→        val job = activeRequests[token] ?: return false
   129→        Log.i(TAG, "Cancelling network request for token: $token")
   130→        job.cancel()
   131→        activeRequests.remove(token)
   132→        return true
   133→    }

   135→    fun shutdown() {
   136→        Log.i(TAG, "Shutting down Runtime Host...")
   137→        try {
   138→            initialized = false
   139→            activeRequests.values.forEach { it.cancel() }
   140→            activeRequests.clear()
   141→
   142→            MangaImageProxy.stop()
   143→
   144→            csPathRegistry.clear()
   145→            csMetadataRegistry.clear()
   146→            aniyomiPathRegistry.clear()
   147→            aniyomiIsAnimeRegistry.clear()
   148→            Log.i(TAG, "Runtime Host shutdown complete")
   149→        } catch (e: Exception) {
   150→            Log.e(TAG, "Error during shutdown: ${e.message}")
   151→        }
   152→    }

   154→    fun getInstalledAnimeExtensions(context: Context, path: String? = null): List<Map<String, Any?>> {
   155→        val extensions = extensionManager(context).fetchInstalledAnimeExtensions(path)

   168→                    "id" to source.id.toString(),

   424→    private data class PrefHandlers(
   425→        val pref: androidx.preference.Preference,
   426→        val click: androidx.preference.Preference.OnPreferenceClickListener?,
   427→        val change: androidx.preference.Preference.OnPreferenceChangeListener?

   432→    @android.annotation.SuppressLint("RestrictedApi")
   433→    fun aniyomiGetPreference(
   434→        context: Context,
   435→        sourceId: String,
   436→        isAnime: Boolean,
   437→    ): List<Map<String, Any?>> = runBlocking {
   438→        sourcePreferences.remove(sourceId)
   439→        
   440→        val prefManager = androidx.preference.PreferenceManager(context)
   441→        prefManager.sharedPreferencesName = "source_$sourceId"
   442→        
   443→        val screen = prefManager.createPreferenceScreen(context)
   444→        
   445→        try {
   446→            media(context, sourceId, isAnime).setupPreferenceScreen(screen)
   447→        } catch (e: com.anymex.runtimehost.aniyomi.NoPreferenceScreenException) {
   448→            return@runBlocking emptyList<Map<String, Any?>>()
   449→        }
   450→
   451→        val list = mutableListOf<Map<String, Any?>>()
   452→        val store = sourcePreferences.getOrPut(sourceId) { mutableMapOf() }
   453→
   454→        fun walk(group: androidx.preference.PreferenceGroup) {
   455→            for (i in 0 until group.preferenceCount) {
   456→                val p = group.getPreference(i)
   457→                store[p.key] = PrefHandlers(p, p.onPreferenceClickListener, p.onPreferenceChangeListener)
   458→
   459→                val map = mutableMapOf(
   460→                    "key" to p.key,
   461→                    "title" to p.title?.toString(),
   462→                    "summary" to p.summary?.toString(),
   463→                    "enabled" to p.isEnabled,
   464→                    "type" to when (p) {
   465→                        is androidx.preference.ListPreference -> "list"
   466→                        is androidx.preference.MultiSelectListPreference -> "multi_select"
   467→                        is androidx.preference.SwitchPreferenceCompat -> "switch"
   468→                        is androidx.preference.EditTextPreference -> "text"
   469→                        is androidx.preference.CheckBoxPreference -> "checkbox"
   470→                        else -> "other"
   471→                    },
   472→                    "value" to when (p) {
   473→                        is androidx.preference.ListPreference -> p.value
   474→                        is androidx.preference.MultiSelectListPreference -> p.values.toList()
   475→                        is androidx.preference.SwitchPreferenceCompat -> p.isChecked
   476→                        is androidx.preference.EditTextPreference -> p.text
   477→                        is androidx.preference.CheckBoxPreference -> p.isChecked
   478→                        else -> null
   479→                    }
   480→                )
   481→
   482→                if (p is androidx.preference.ListPreference) {
   483→                    map["entries"] = p.entries?.map { it.toString() }
   484→                    map["entryValues"] = p.entryValues?.map { it.toString() }
   485→                } else if (p is androidx.preference.MultiSelectListPreference) {
   486→                    map["entries"] = p.entries?.map { it.toString() }
   487→                    map["entryValues"] = p.entryValues?.map { it.toString() }
   488→                }
   489→
   490→                list += map
   491→                if (p is androidx.preference.PreferenceCategory) walk(p)
   492→            }
   493→        }
   494→
   495→        walk(screen)
   496→        list
   497→    }

   499→    fun aniyomiSavePreference(
   500→        context: Context,
   501→        sourceId: String,
   502→        key: String,
   503→        action: String?,
   504→        value: Any?,
   505→    ): Boolean = runBlocking(Dispatchers.Main) {
   506→        val isAnime = aniyomiIsAnimeRegistry[sourceId] ?: true
   507→        val handler = sourcePreferences[sourceId]?.get(key) ?: return@runBlocking false
   508→        val pref = handler.pref
   509→        
   510→        val actualAction = action ?: "change"
   511→
   512→        val actualValue = if (pref is androidx.preference.MultiSelectListPreference && value is List<*>) {
   513→            value.filterIsInstance<String>().toSet()
   514→        } else {
   515→            value
   516→        }
   517→
   518→        if (actualAction == "click") {
   519→            handler.click?.onPreferenceClick(pref)
   520→        } else {
   521→            handler.change?.onPreferenceChange(pref, actualValue)
   522→        }
   523→
   524→        val prefs = context.getSharedPreferences("source_$sourceId", Context.MODE_PRIVATE)
   525→        val editor = prefs.edit()
   526→
   527→        when (pref) {
   528→            is androidx.preference.SwitchPreferenceCompat -> {
   529→                val b = value as Boolean
   530→                pref.isChecked = b
   531→                editor.putBoolean(key, b)
   532→            }
   533→            is androidx.preference.ListPreference -> {
   534→                val s = value as String
   535→                pref.value = s
   536→                editor.putString(key, s)
   537→            }
   538→            is androidx.preference.EditTextPreference -> {
   539→                val s = value as String
   540→                pref.text = s
   541→                editor.putString(key, s)
   542→            }
   543→            is androidx.preference.MultiSelectListPreference -> {
   544→                val newSet = when (value) {
   545→                    is List<*> -> value.filterIsInstance<String>().toSet()
   546→                    is Set<*> -> value.filterIsInstance<String>().toSet()
   547→                    else -> emptySet()
   548→                }
   549→                pref.values = newSet.toMutableSet()
   550→                editor.putStringSet(key, newSet)
   551→            }
   552→            is androidx.preference.CheckBoxPreference -> {
   553→                val b = value as Boolean
   554→                pref.isChecked = b
   555→                editor.putBoolean(key, b)
   556→            }
   557→        }
   558→        editor.apply()
   559→
   560→        true
   561→    }

   563→    fun csLoadPlugin(context: Context, path: String): Boolean {
   564→        val file = File(path)
   565→        if (!file.exists()) {
   566→            Log.e(TAG, "Plugin file not found: $path")
   567→            return false
   568→        }
   569→        
   570→        return runBlocking {
   571→            val success = PluginManager.loadPlugin(context, file)
   572→            if (success) {
   573→                val pluginData = PluginManager.getPluginsOnline()
   574→                APIHolder.apis.forEach { provider ->
   575→                    if (provider.sourcePlugin == path || provider.sourcePlugin?.contains(file.name) == true) {
   576→                        val data = pluginData.firstOrNull { it.filePath == provider.sourcePlugin }
   577→                        val pluginInstance = PluginManager.plugins[provider.sourcePlugin]
   578→                        val hasSettings = (pluginInstance as? com.lagradost.cloudstream3.plugins.Plugin)?.openSettings != null
   579→                        csPathRegistry[provider.name] = path
   580→                        csMetadataRegistry[provider.name] = ProviderMetadata(
   581→                            id = provider.name,
   582→                            name = provider.name,
   583→                            mainUrl = provider.mainUrl,
   584→                            lang = provider.lang,
   585→                            sourcePlugin = provider.sourcePlugin,
   586→                            iconUrl = data?.iconUrl,
   587→                            internalName = data?.internalName,
   588→                            hasSettings = hasSettings
   589→                        )
   590→                        Log.d(TAG, "Registered CloudStream Provider: ${provider.name}")
   591→                    }
   592→                }
   593→            }
   594→            success
   595→        }
   596→    }

   598→    fun csUnloadPlugin(apiName: String): Boolean {
   599→        if (apiName.isBlank()) return false
   600→        
   601→        val provider = APIHolder.apis.find { it.name.equals(apiName, ignoreCase = true) }
   602→        val internalName = provider?.sourcePlugin ?: csMetadataRegistry[apiName]?.mainUrl ?: apiName
   603→        
   604→        Log.i(TAG, "Attempting to unload plugin: $apiName (Internal Path/ID: $internalName)")
   605→
   606→        val removedCount = APIHolder.apis.count { 
   607→            it.name.equals(apiName, ignoreCase = true) || 
   608→            (it.sourcePlugin != null && it.sourcePlugin!!.endsWith(internalName, ignoreCase = true)) 
   609→        }
   610→
   611→        APIHolder.apis.removeIf { 
   612→            it.name.equals(apiName, ignoreCase = true) || 
   613→            (it.sourcePlugin != null && it.sourcePlugin!!.endsWith(internalName, ignoreCase = true)) 
   614→        }
   615→
   616→        csMetadataRegistry.remove(apiName)
   617→        
   618→        val fileName = internalName.substringAfterLast("/")
   619→        if (fileName.length > 3) {
   620→            val extraRemoved = APIHolder.apis.count { it.sourcePlugin?.endsWith(fileName, ignoreCase = true) == true }
   621→            APIHolder.apis.removeIf { it.sourcePlugin?.endsWith(fileName, ignoreCase = true) == true }
   622→            Log.d(TAG, "Extra removal check for '$fileName' hit $extraRemoved providers")
   623→        }
   624→
   625→        PluginManager.unloadPlugin(internalName)
   626→        Log.i(TAG, "Unloaded $removedCount providers for plugin: $apiName")
   627→        return true
   628→    }

   875→    private fun parseEpisodeInfoFromName(name: String): Pair<Float, Int?> {
   876→        val seasonRegex = Regex("""(?i)\b(?:s|season)\s*(\d+)""")
   877→        val seasonMatch = seasonRegex.find(name)
   878→        val season = seasonMatch?.groupValues?.get(1)?.toIntOrNull()
   879→
   880→        val cleanName = if (seasonMatch != null) name.replace(seasonMatch.value, "") else name
   881→
   882→        val prefixRegex = Regex("""(?i)(?:chapter|ch\.|ch|ep\.|ep|episode|e)\s*(\d+(\.\d+)?)""")
   883→        val prefixMatch = prefixRegex.find(cleanName)
   884→        if (prefixMatch != null) {
   885→            val epVal = prefixMatch.groupValues[1].toFloatOrNull() ?: -1f
   886→            return Pair(epVal, season)
   887→        }
   888→
   889→        val fallbackRegex = Regex("""(\d+(\.\d+)?)""")
   890→        val epVal = fallbackRegex.find(cleanName)?.groupValues?.get(1)?.toFloatOrNull() ?: -1f
   891→        return Pair(epVal, season)
   892→    }
