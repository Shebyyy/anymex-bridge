   487→   487→function baseName(name){
   488→   488→  return (name||'').replace(/\s*[\-–—(].*$/, '').replace(/\s+(All|Multi|English|Japanese|Spanish|Portuguese|French|Italian|German|Arabic|Hindi|Thai|Vietnamese|Indonesian|Korean|Chinese|Turkish|Polish|Russian)$/i, '').trim();

   491→   491→function populateSourceDropdown(){
   492→   492→  const sel=document.getElementById('invokeSource');
   493→   493→  if(!_sources.length){
   494→   494→    sel.innerHTML='<option value="">-- No sources loaded --</option>';
   495→   495→    return;
   496→   496→  }
   497→   497→  // Group by type, then by base name
   498→   498→  const byType={};
   499→   499→  _sources.forEach(s=>{
   500→   500→    const t=s.type||'other';
   501→   501→    if(!byType[t])byType[t]=[];
   502→   502→    byType[t].push(s);
   503→   503→  });
   504→   504→  let html='<option value="">-- Select a source --</option>';
   505→   505→  for(const[type,sources] of Object.entries(byType)){
   506→   506→    html+=`<optgroup label="${type}">`;
   507→   507→    // Group by base name
   508→   508→    const groups=new Map();
   509→   509→    sources.forEach(s=>{
   510→   510→      const bn=baseName(s.name);
   511→   511→      if(!groups.has(bn))groups.set(bn,[]);
   512→   512→      groups.get(bn).push(s);
   513→   513→    });
   514→   514→    // Sort base names
   515→   515→    [...groups.keys()].sort().forEach(bn=>{

   624→   624→    }
