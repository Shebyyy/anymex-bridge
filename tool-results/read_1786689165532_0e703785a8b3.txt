    50→  List<Map<String, dynamic>> get _displayManagers {
    51→    final List<Map<String, dynamic>> list = [];
    52→    final activeManagers = em.managers;
    53→
    54→    for (final m in activeManagers) {
    55→      // No iOS restriction — plugin managers are now supported on iOS via JNI bridge
    56→      list.add({
    57→        'name': m.name,
    58→        'manager': m,
    59→        'isMock': false,
    60→      });
    61→    }
    62→
    63→    // Show mock entries for Aniyomi/CloudStream on all platforms (including iOS)
    64→    final hasAniyomi = activeManagers.any((m) => m.name.toLowerCase().contains('aniyomi'));
    65→    final hasCloudStream = activeManagers.any((m) => m.name.toLowerCase().contains('cloudstream'));
    66→
    67→    if (!hasAniyomi) {
    68→      list.add({
    69→        'name': 'Aniyomi',
    70→        'manager': null,
    71→        'isMock': true,
    72→      });
    73→    }
    74→    if (!hasCloudStream) {
    75→      list.add({
    76→        'name': 'CloudStream',
    77→        'manager': null,
    78→        'isMock': true,
    79→      });
    80→    }
    81→
    82→    list.sort((a, b) {
    83→      final managerA = a['manager'] as Extension?;
    84→      final managerB = b['manager'] as Extension?;
    85→
    86→      final aRequires = managerA?.requiresPlugin ?? _isMockPluginRequired(a['name'] as String);
    87→      final bRequires = managerB?.requiresPlugin ?? _isMockPluginRequired(b['name'] as String);
    88→
    89→      if (aRequires != bRequires) {
    90→        return aRequires ? 1 : -1;
    91→      }
    92→
    93→      final aName = (a['name'] as String).toLowerCase();
    94→      final bName = (b['name'] as String).toLowerCase();
    95→      if (aName.contains('mangayomi')) return -1;
    96→      if (bName.contains('mangayomi')) return 1;
    97→
    98→      return aName.compareTo(bName);
    99→    });
   100→
   101→    return list;
   102→  }

   742→  Widget? _buildFab() {
   743→    if (_managerIndex >= _displayManagers.length) return null;
   744→    final selectedManagerData = _displayManagers[_managerIndex];

   746→
   747→    final manager = selectedManagerData['manager'] as Extension;
   748→    final supported = _tab == ItemType.anime
   749→        ? manager.supportsAnime
   750→        : _tab == ItemType.manga

   752→            : manager.supportsNovel;

   754→    final colors = context.colors;

   780→  String _path(String url) {
   781→    try {
   782→      final p = Uri.parse(url).path;

   784→    } catch (_) {
   785→      return url;
