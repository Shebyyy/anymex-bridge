    26→class _SettingsExtensionManagerState extends State<SettingsExtensionManager> {
    27→  final _pluginManager = PluginManager();
    28→  bool _isCheckingUpdate = false;
    29→  bool _isSyncingLocalApk = false;
    30→  bool _needsRestart = false;
    31→  final RxBool _isLoadedFromStorage = false.obs;
    32→  String get _installedVersion => AnymeXRuntimeBridge.installedVersion;
    33→
    34→  String get _installedReleaseTitle =>
    35→      AnymeXRuntimeBridge.installedReleaseTitle;
    36→
    37→  bool get _isPluginInstalled => AnymeXRuntimeBridge.isPluginInstalled;
    38→
    39→  @override
    40→  void initState() {
    41→    super.initState();
    42→    _checkStorageStatus();
    43→  }
    44→
    45→  void _checkStorageStatus() async {
    46→    final status = await AnymeXRuntimeBridge.isLoadedFromStorage();
    47→    _isLoadedFromStorage.value = status;
    48→  }
    49→
    50→  void _showInstallPopup() async {
    51→    final oldVersion = _installedVersion;
    52→    await _pluginManager.showInstallSheet(context);
    53→    if (mounted) {
    54→      if (_installedVersion != oldVersion) {
    55→        setState(() {
    56→          _needsRestart = true;
    57→        });
    58→      } else {
    59→        setState(() {});
    60→      }
    61→    }
    62→  }
    63→
    64→  void _showUpdatePopup() async {
    65→    final release = await _pluginManager.fetchLatestRelease();
    66→    if (!mounted) return;
    67→    if (release == null) {
    68→      errorSnackBar('Failed to check for updates.');
    69→      return;
    70→    }
    71→    final currentVersion = _installedVersion;
    72→    if (currentVersion.isEmpty) {
    73→      _showInstallPopup();
    74→      return;
    75→    }
    76→    if (_pluginManager.isNewerVersion(currentVersion, release.tagName)) {
    77→      final oldVersion = _installedVersion;
    78→      await _pluginManager.showUpdateSheet(
    79→        context,
    80→        release: release,
    81→        installedVersion: currentVersion,
    82→      );
    83→      if (mounted) {
    84→        if (_installedVersion != oldVersion) {
    85→          setState(() {
    86→            _needsRestart = true;
    87→          });
    88→        } else {
    89→          setState(() {});
    90→        }
    91→      }
    92→    } else {
    93→      successSnackBar('Plugin is already up to date.');
    94→    }
    95→  }
    96→
    97→  void _checkForUpdates() async {
    98→    setState(() => _isCheckingUpdate = true);
    99→    try {
   100→      final release = await _pluginManager.fetchLatestRelease();
   101→      if (!mounted) return;
   102→      if (release == null) {
   103→        errorSnackBar('Failed to check for updates.');
   104→        return;
   105→      }
   106→      final currentVersion = _installedVersion;
   107→      if (currentVersion.isEmpty) {
   108→        _showInstallPopup();
   109→        return;
   110→      }
   111→      if (_pluginManager.isNewerVersion(currentVersion, release.tagName)) {
   112→        _showUpdatePopup();
   113→      } else {
   114→        successSnackBar('Plugin is already up to date.');
   115→      }
   116→    } finally {
   117→      if (mounted) setState(() => _isCheckingUpdate = false);
   118→    }
   119→  }
   120→
   121→  void _syncLocalApk() async {
   122→    if (_isSyncingLocalApk) return;
   123→
   124→    final result = await FilePicker.platform.pickFiles(
   125→      type: FileType.custom,
   126→      allowedExtensions: Platform.isAndroid ? const ['apk'] : const ['jar'],
   127→      allowMultiple: false,
   128→      withData: false,
   129→    );
   130→
   131→    final apkPath = result?.files.single.path;
   132→    if (apkPath == null || apkPath.isEmpty) return;
   133→    if (!mounted) return;
   134→
   135→    setState(() => _isSyncingLocalApk = true);
   136→    try {
   137→      final oldVersion = _installedVersion;
   138→      final synced = await _pluginManager.syncLocalApk(apkPath);
   139→      if (mounted && synced) {
   140→        _checkStorageStatus();
   141→        if (_installedVersion != oldVersion) {
   142→          setState(() {
   143→            _needsRestart = true;
   144→          });
   145→        } else {
   146→          setState(() {});
   147→        }
   148→      }
   149→    } finally {
   150→      if (mounted) setState(() => _isSyncingLocalApk = false);
   151→    }
   152→  }
   153→
   154→  void _forceReDownload() async {
   155→    final bridge = AnymeXRuntimeBridge.controller;
   156→    if (bridge.isDownloading.value) return;
   157→    try {
   158→      await AnymeXRuntimeBridge.setupRuntime(force: true);
   159→      if (bridge.isReady.value) {
   160→        await Get.find<ExtensionManager>()
   161→            .onRuntimeBridgeInitialization(force: true);
   162→        _checkStorageStatus();
   163→        if (mounted) {
   164→          setState(() {
   165→            _needsRestart = true;
   166→          });
   167→          successSnackBar('Plugin re-downloaded successfully. Please restart to apply.');
   168→        }
   169→      }
   170→    } catch (error) {
   171→      if (mounted) errorSnackBar('Re-download failed: $error');
   172→    }
   173→  }
   174→

   286→  void _performRollback(PluginRelease release) async {
   287→    final bridge = AnymeXRuntimeBridge.controller;
   288→    if (bridge.isDownloading.value) return;
   289→    try {
   290→      await AnymeXRuntimeBridge.setupRuntime(
   291→        customDownloadUrl: release.asset.downloadUrl,
   292→        force: true,
   293→      );
   294→      if (bridge.isReady.value) {
   295→        await Get.find<ExtensionManager>()
   296→            .onRuntimeBridgeInitialization(force: true);
   297→        _pluginManager.persistInstalledRelease(release);
   298→        _checkStorageStatus();
   299→        if (mounted) {
   300→          setState(() {
   301→            _needsRestart = true;
   302→          });
   303→          successSnackBar('Rollback to ${release.tagName} successful. Restart app to apply.');
   304→        }
   305→      }
   306→    } catch (error) {
   307→      if (mounted) errorSnackBar('Rollback failed: $error');
   308→    }
   309→  }

   689→  List<Widget> _buildExtensionSettings(BuildContext context) {
   690→    final colors = Theme.of(context).colorScheme;
   691→    final em = Get.find<ExtensionManager>();
   692→    final settingsList = <Widget>[];
   693→
   694→    for (final manager in em.managers) {
   695→      final managerSettings = manager.settings;
   696→      if (managerSettings == null || managerSettings.isEmpty) continue;
   697→
   698→      settingsList.add(const SizedBox(height: 25));
   699→      settingsList.add(
   700→        Padding(
   701→          padding: const EdgeInsets.only(left: 4.0, bottom: 10.0),
   702→          child: Text(
   703→            '${manager.name.toUpperCase()} SETTINGS',
   704→            style: TextStyle(
   705→              fontSize: 12,
   706→              fontWeight: FontWeight.bold,
   707→              color: colors.primary,
   708→              letterSpacing: 1.2,
   709→            ),
   710→          ),
   711→        ),
   712→      );
   713→
   714→      final children = <Widget>[];
   715→      for (final entry in managerSettings.entries) {
   716→        final setting = entry.value;
   717→
   718→        if (setting.type == 'bool') {
   719→          children.add(
   720→            CustomSwitchTile(
   721→              icon: Icons.settings_input_component_rounded,
   722→              title: setting.label,
   723→              description: setting.description,
   724→              switchValue: setting.value as bool,
   725→              onChanged: (val) {
   726→                setting.onChanged(val);
   727→                setState(() {});
   728→              },
   729→            ),
   730→          );
   731→        } else if (setting.type == 'string') {
   732→          children.add(
   733→            CustomTile(
   734→              icon: Icons.folder_open_rounded,
   735→              title: setting.label,
   736→              description: (setting.value as String).isNotEmpty
   737→                  ? setting.value as String
   738→                  : setting.description,
   739→              onTap: () => _showTextInputDialog(
   740→                context,
   741→                title: setting.label,
   742→                initialValue: setting.value as String,
   743→                onSave: (val) {
   744→                  setting.onChanged(val);
   745→                  setState(() {});
   746→                },
   747→              ),
   748→            ),
   749→          );
   750→        }
   751→      }
   752→
   753→      if (children.isNotEmpty) {
   754→        settingsList.add(
   755→          Container(
   756→            decoration: BoxDecoration(
   757→              color: colors.surfaceContainer.withValues(alpha: 0.3),
   758→              borderRadius: BorderRadius.circular(18),
   759→              border: Border.all(
   760→                  color: colors.outlineVariant.withValues(alpha: 0.4)),
   761→            ),
   762→            child: Column(children: children),
   763→          ),
   764→        );
   765→      }
   766→    }
   767→
   768→    return settingsList;
   769→  }

   771→  void _showTextInputDialog(
   772→    BuildContext context, {
   773→    required String title,
   774→    required String initialValue,
   775→    required ValueChanged<String> onSave,
   776→  }) {
   777→    final controller = TextEditingController(text: initialValue);
   778→    showDialog(
   779→      context: context,
   780→      builder: (context) {
   781→        return AlertDialog(
   782→          title: Text(title),
   783→          content: TextField(
   784→            controller: controller,
   785→            decoration: const InputDecoration(
   786→              border: OutlineInputBorder(),
   787→              labelText: 'Enter value',
   788→            ),
   789→          ),
   790→          actions: [
   791→            TextButton(
   792→              onPressed: () => Navigator.of(context).pop(),
   793→              child: const Text('Cancel'),
   794→            ),
   795→            TextButton(
   796→              onPressed: () {
   797→                onSave(controller.text);
   798→                Navigator.of(context).pop();
   799→              },
   800→              child: const Text('Save'),
   801→            ),
   802→          ],
   803→        );
   804→      },
   805→    );
   806→  }
