    27→class _SettingsExtensionManagerState extends State<SettingsExtensionManager> {
    28→  final _pluginManager = PluginManager();
    29→  bool _isCheckingUpdate = false;
    30→  bool _isSyncingLocalApk = false;
    31→  bool _needsRestart = false;
    32→  final RxBool _isLoadedFromStorage = false.obs;
    33→  String get _installedVersion => AnymeXRuntimeBridge.installedVersion;
    34→
    35→  String get _installedReleaseTitle =>

    40→  @override
    41→  void initState() {
    42→    super.initState();
    43→    _checkStorageStatus();

    45→
    46→  void _checkStorageStatus() async {
    47→    final status = await AnymeXRuntimeBridge.isLoadedFromStorage();
    48→    _isLoadedFromStorage.value = status;
    49→  }
    50→
    51→  void _showInstallPopup() async {
    52→    final oldVersion = _installedVersion;
    53→    await _pluginManager.showInstallSheet(context);
    54→    if (mounted) {
    55→      if (_installedVersion != oldVersion) {
    56→        setState(() {
    57→          _needsRestart = true;
    58→        });
    59→      } else {
    60→        setState(() {});
    61→      }
    62→    }
    63→  }
    64→
    65→  void _showUpdatePopup() async {
    66→    final release = await _pluginManager.fetchLatestRelease();
    67→    if (!mounted) return;
    68→    if (release == null) {
    69→      errorSnackBar('Failed to check for updates.');
    70→      return;
    71→    }
    72→    final currentVersion = _installedVersion;
    73→    if (currentVersion.isEmpty) {
    74→      _showInstallPopup();
    75→      return;
    76→    }
    77→    if (_pluginManager.isNewerVersion(currentVersion, release.tagName)) {
    78→      final oldVersion = _installedVersion;
    79→      await _pluginManager.showUpdateSheet(
    80→        context,
    81→        release: release,
    82→        installedVersion: currentVersion,
    83→      );
    84→      if (mounted) {
    85→        if (_installedVersion != oldVersion) {
    86→          setState(() {
    87→            _needsRestart = true;
    88→          });
    89→        } else {
    90→          setState(() {});
    91→        }
    92→      }
    93→    } else {
    94→      successSnackBar('Plugin is already up to date.');
    95→    }
    96→  }
    97→
    98→  void _checkForUpdates() async {
    99→    setState(() => _isCheckingUpdate = true);
   100→    try {
   101→      final release = await _pluginManager.fetchLatestRelease();
   102→      if (!mounted) return;
   103→      if (release == null) {
   104→        errorSnackBar('Failed to check for updates.');
   105→        return;
   106→      }
   107→      final currentVersion = _installedVersion;
   108→      if (currentVersion.isEmpty) {
   109→        _showInstallPopup();
   110→        return;
   111→      }
   112→      if (_pluginManager.isNewerVersion(currentVersion, release.tagName)) {
   113→        _showUpdatePopup();
   114→      } else {
   115→        successSnackBar('Plugin is already up to date.');
   116→      }
   117→    } finally {
   118→      if (mounted) setState(() => _isCheckingUpdate = false);
   119→    }
   120→  }
   121→
   122→  void _syncLocalApk() async {
   123→    if (_isSyncingLocalApk) return;
   124→
   125→    final result = await FilePicker.platform.pickFiles(
   126→      type: FileType.custom,
   127→      allowedExtensions: Platform.isAndroid ? const ['apk'] : const ['jar'],
   128→      allowMultiple: false,
   129→      withData: false,
   130→    );
   131→
   132→    final apkPath = result?.files.single.path;
   133→    if (apkPath == null || apkPath.isEmpty) return;
   134→    if (!mounted) return;
   135→
   136→    setState(() => _isSyncingLocalApk = true);
   137→    try {
   138→      final oldVersion = _installedVersion;
   139→      final synced = await _pluginManager.syncLocalApk(apkPath);
   140→      if (mounted && synced) {
   141→        _checkStorageStatus();
   142→        if (_installedVersion != oldVersion) {
   143→          setState(() {
   144→            _needsRestart = true;
   145→          });
   146→        } else {
   147→          setState(() {});
   148→        }
   149→      }
   150→    } finally {
   151→      if (mounted) setState(() => _isSyncingLocalApk = false);
   152→    }
   153→  }
   154→
   155→  void _forceReDownload() async {
   156→    final bridge = AnymeXRuntimeBridge.controller;
   157→    if (bridge.isDownloading.value) return;
   158→    try {
   159→      await AnymeXRuntimeBridge.setupRuntime(force: true);
   160→      if (bridge.isReady.value) {
   161→        await Get.find<ExtensionManager>()
   162→            .onRuntimeBridgeInitialization(force: true);
   163→        _checkStorageStatus();
   164→        if (mounted) {
   165→          setState(() {
   166→            _needsRestart = true;
   167→          });
   168→          successSnackBar(
   169→              'Plugin re-downloaded successfully. Please restart to apply.');
   170→        }
   171→      }
   172→    } catch (error) {
   173→      if (mounted) errorSnackBar('Re-download failed: $error');
   174→    }
   175→  }

   294→  void _performRollback(PluginRelease release) async {
   295→    final bridge = AnymeXRuntimeBridge.controller;
   296→    if (bridge.isDownloading.value) return;
   297→    try {
   298→      await AnymeXRuntimeBridge.setupRuntime(
   299→        customDownloadUrl: release.asset.downloadUrl,
   300→        force: true,
   301→      );
   302→      if (bridge.isReady.value) {
   303→        await Get.find<ExtensionManager>()
   304→            .onRuntimeBridgeInitialization(force: true);
   305→        _pluginManager.persistInstalledRelease(release);
   306→        _checkStorageStatus();
   307→        if (mounted) {
   308→          setState(() {
   309→            _needsRestart = true;
   310→          });
   311→          successSnackBar(
   312→              'Rollback to ${release.tagName} successful. Restart app to apply.');
   313→        }
   314→      }
   315→    } catch (error) {
   316→      if (mounted) errorSnackBar('Rollback failed: $error');
   317→    }
   318→  }

   942→  List<Widget> _buildExtensionSettings(BuildContext context) {
   943→    final colors = Theme.of(context).colorScheme;
   944→    final em = Get.find<ExtensionManager>();
   945→    final settingsList = <Widget>[];
   946→
   947→    for (final manager in em.managers) {
   948→      final managerSettings = manager.settings;
   949→      if (managerSettings == null || managerSettings.isEmpty) continue;
   950→
   951→      settingsList.add(const SizedBox(height: 25));
   952→      settingsList.add(
   953→        Padding(
   954→          padding: const EdgeInsets.only(left: 4.0, bottom: 10.0),
   955→          child: Text(
   956→            '${manager.name.toUpperCase()} SETTINGS',
   957→            style: TextStyle(
   958→              fontSize: 12,
   959→              fontWeight: FontWeight.bold,
   960→              color: colors.primary,
   961→              letterSpacing: 1.2,
   962→            ),
   963→          ),
   964→        ),
   965→      );
   966→
   967→      final children = <Widget>[];
   968→      for (final entry in managerSettings.entries) {
   969→        final setting = entry.value;
   970→
   971→        if (setting.type == 'bool') {
   972→          children.add(
   973→            CustomSwitchTile(
   974→              icon: Icons.settings_input_component_rounded,
   975→              title: setting.label,
   976→              description: setting.description,
   977→              switchValue: setting.value as bool,
   978→              onChanged: (val) {
   979→                setting.onChanged(val);
   980→                setState(() {});
   981→              },
   982→            ),
   983→          );
   984→        } else if (setting.type == 'string') {
   985→          children.add(
   986→            CustomTile(
   987→              icon: Icons.folder_open_rounded,
   988→              title: setting.label,
   989→              description: (setting.value as String).isNotEmpty
   990→                  ? setting.value as String
   991→                  : setting.description,
   992→              onTap: () => _showTextInputDialog(
   993→                context,
   994→                title: setting.label,
   995→                initialValue: setting.value as String,
   996→                onSave: (val) {
   997→                  setting.onChanged(val);
   998→                  setState(() {});
   999→                },
  1000→              ),
  1001→            ),
  1002→          );
  1003→        }
  1004→      }
  1005→
  1006→      if (children.isNotEmpty) {
  1007→        settingsList.add(
  1008→          Container(
  1009→            decoration: BoxDecoration(
  1010→              color: colors.surfaceContainer.withValues(alpha: 0.3),
  1011→              borderRadius: BorderRadius.circular(18),
  1012→              border: Border.all(
  1013→                  color: colors.outlineVariant.withValues(alpha: 0.4)),
  1014→            ),
  1015→            child: Column(children: children),
  1016→          ),
  1017→        );
  1018→      }
  1019→    }
  1020→
  1021→    return settingsList;
  1022→  }
