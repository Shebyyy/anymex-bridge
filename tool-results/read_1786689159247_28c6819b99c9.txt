    50→    50→  List<Map<String, dynamic>> get _displayManagers {
    51→    51→    final List<Map<String, dynamic>> list = [];
    52→    52→    final activeManagers = em.managers;
    53→    53→
    54→    54→    for (final m in activeManagers) {
    55→    55→      // No iOS restriction — plugin managers are now supported on iOS via JNI bridge
    56→    56→      list.add({
    57→    57→        'name': m.name,
    58→    58→        'manager': m,
    59→    59→        'isMock': false,
    60→    60→      });
    61→    61→    }
    62→    62→
    63→    63→    // Show mock entries for Aniyomi/CloudStream on all platforms (including iOS)
    64→    64→    final hasAniyomi = activeManagers.any((m) => m.name.toLowerCase().contains('aniyomi'));
    65→    65→    final hasCloudStream = activeManagers.any((m) => m.name.toLowerCase().contains('cloudstream'));
    66→    66→
    67→    67→    if (!hasAniyomi) {
    68→    68→      list.add({
    69→    69→        'name': 'Aniyomi',
    70→    70→        'manager': null,
    71→    71→        'isMock': true,
    72→    72→      });
    73→    73→    }
    74→    74→    if (!hasCloudStream) {
    75→    75→      list.add({
    76→    76→        'name': 'CloudStream',
    77→    77→        'manager': null,
    78→    78→        'isMock': true,
    79→    79→      });
    80→    80→    }
    81→    81→
    82→    82→    list.sort((a, b) {
    83→    83→      final managerA = a['manager'] as Extension?;
    84→    84→      final managerB = b['manager'] as Extension?;
    85→    85→
    86→    86→      final aRequires = managerA?.requiresPlugin ?? _isMockPluginRequired(a['name'] as String);
    87→    87→      final bRequires = managerB?.requiresPlugin ?? _isMockPluginRequired(b['name'] as String);
    88→    88→
    89→    89→      if (aRequires != bRequires) {
    90→    90→        return aRequires ? 1 : -1;
    91→    91→      }
    92→    92→
    93→    93→      final aName = (a['name'] as String).toLowerCase();
    94→    94→      final bName = (b['name'] as String).toLowerCase();
    95→    95→      if (aName.contains('mangayomi')) return -1;
    96→    96→      if (bName.contains('mangayomi')) return 1;
    97→    97→
    98→    98→      return aName.compareTo(bName);
    99→    99→    });
   100→   100→
   101→   101→    return list;
   102→   102→  }

   104→   104→  bool _isMockPluginRequired(String name) {

   110→   110→  Future<void> _addRepos(ItemType type, List<String> urls) async {
   111→   111→    final currentManagerData = _displayManagers[_managerIndex];
   112→   112→    if (currentManagerData['isMock'] == true) return;
   113→   113→    final activeManager = currentManagerData['manager'] as Extension;
   114→   114→    await em.addRepos(urls, type, activeManager.id);
   115→   115→    widget.onSave?.call();
   116→   116→    if (mounted) setState(() {});
   117→   117→  }

   119→   119→  Future<void> _removeRepo(Repo repo, ItemType type) async {
   120→   120→    final key = '${repo.url}_${type.name}';
   121→   121→    setState(() => _deleting[key] = true);
   122→   122→    try {
   123→   123→      await em.removeRepo(repo, type);
   124→   124→      widget.onSave?.call();
   125→   125→    } catch (_) {
   126→   126→      snackBar('Failed to remove repo');
   127→   127→    } finally {
   128→   128→      if (mounted) setState(() => _deleting.remove(key));
   129→   129→    }
   130→   130→  }
